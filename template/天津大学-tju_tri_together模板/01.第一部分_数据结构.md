## 1.1 HEAP(BY LIYUAN)
```cpp
// 最小堆
# include<cstdio>
const int MAX=200000;
int len;            // 堆的长度
int a[MAX];
void up(int a[], int p) {   // p 为调整元素的下标
    int tmp = a[p];
    while(p >= 2 && a[p>>1] > tmp) {
        a[p] = a[p>>1];
        p >>= 1;
    }
    a[p] = tmp;
}

void down(int a[], int p) {
    int x = p << 1, tmp = a[p];
    while(x <= len) {
        if(x + 1 <= len && a[x] > a[x+1]) x++;
        if(tmp <= a[x])break;
        a[p] = a[x];
        p = x;
        x <<= 1;
    }
    a[p] = tmp;
}

void push(int a[], int v) {
    a[++len] = v;
    up(a, len);
}

int pop(int a[]) {
    int tmp = a[1];
    if(len==0)return -1;
    a[1] = a[len--];
    if(len==0)return tmp;
    down(a, 1);
    return tmp;
}
int main() {
    int n,i,s;
    char order[10];
    while(scanf("%d", &n) && n) {
        len = 0;
        for(i = 1; i <= n; i++) {
            scanf("%s", order);
            if(order[0] == 'B') {
                scanf("%d", &s);
                push(a, s);
            } else {
                printf("%d\n", pop(a));
            }
        }
    }
    return 0;
}
```

```cpp
#include <cstdio>

#include <string.h>

    INIT();
    return 0;

    int n,d;
    scanf("%d", &n);
    memset(c, 0, sizeof(c));
    for(int i = 1; i <= n; i++) {
        scanf("%d", &d);
        Change(i, d);
    }
    return;
}
int lowbit(int k) {
    return k & (-k);
}
int Sum(int k) {
    int sum = 0;
    while(k > 0) {
        sum += c[k];
        k -= lowbit(k);
    }
    return sum;
}
void Change(int k, int d) {
    while(k < M) {
        c[k] += d;
        k += lowbit(k);
    }
    return;
}
```

```cpp
            x = y = z = 0;

        }
    }
    return;

    int i,j,k;
    for(i = x; i < M; i += lowbit(i)) {
        for(j = y; j < M; j += lowbit(j)) {
            for(k = z; k < M; k += lowbit(k)) {
                c[i][j][k] += d;
            }
        }
    }
    return;

    for(i = x; i > 0; i -= lowbit(i)) {
        for(j = y; j > 0; j -= lowbit(j)) {
            for(k = z; k > 0; k -= lowbit(k)) {
                r += c[i][j][k];
            }
        }
    }
```

```cpp
// n 个 m维图形的并（线段求长度并，矩形求面积并，立方体则是体积并，以此类推）

#include <iostream>
#include <cstdio>
#include <string.h>
using namespace std;
typedef int (*Rec)[2];
typedef long long LL;
const int M = 15;           // 最多14维

void addrec(Rec rec);

        init();

    }
    return 0;
}

    res = 0;
    top = 0;
    for (int i = 0; i < n; i++) {
        for (int j = 1; j <= m; j++) {
            scanf("%d", &tmp[j][0]);
        }
        for (int j = 1; j <= m; j++) {
            scanf("%d", &tmp[j][1]);
        }
        for (int j = 1;j <= m;j++) {
            if (tmp[j][0] > tmp[j][1]) 
                swap(tmp[j][0], tmp[j][1]);
        }
        addrec(tmp);
    }
    return;
}

void sol() {
    for (int i = 0;i < top;i++) {
        res = (res + calrec(pool[i])) % MOD;
    }
    printf("%d\n", res);
    return;
}

void addrec(Rec rec) {
    int topp;
    for (int i = 0; i < top; i++) {
        copyrec(pool[i], tp[i]);
    }
    topp = top;
    top = 0;
    for (int i = 0; i < topp; i++) {
        if(cross(tp[i], rec)) {
            split(tp[i], rec);
        } else {
            copyrec(tp[i], pool[top++]);
        }
    }
    copyrec(rec, pool[top++]);
    return;
}

bool cross(Rec a, Rec b) {
    // 判断两个m维矩形是否相交
    for (int i = 1; i <= m; i++) {
        if (a[i][0] > b[i][1] )
            return false;
        if (a[i][1] < b[i][0])
            return false;
    }
    return true;
}

void split(Rec a, Rec b) {
    int temp[M][2];
    copyrec(a, temp);
    for (int i = 1; i <= m; i++) {
        if (a[i][0] < b[i][0]) {
            temp[i][1] = b[i][0];
            copyrec(temp, pool[top++]);
            temp[i][1] = a[i][1];
            a[i][0] = b[i][0];
        }
        if (a[i][1] > b[i][1]) {
            temp[i][0] = b[i][1];
            copyrec(temp, pool[top++]);
            temp[i][0] = a[i][0];
            a[i][1] = b[i][1];
        }
        temp[i][0] = b[i][0];
        temp[i][1] = b[i][1];
    }
    return;
}        

int calrec(Rec a) {
    LL ans = 1;
    for (int i = 1;i <= m;i++) {
        ans = (ans * (a[i][1] - a[i][0])) % MOD;
    }
    return (int) ans;
}

void copyrec(Rec a, Rec b) {
    for (int i = 1;i <= m;i++) {
        b[i][0] = a[i][0];
        b[i][1] = a[i][1];
    }
    return;
}
```

## 1.7 划分树求和
```cpp
// tju 3556 hdu 3473
// 给出一个 n 个数的数列，m 个查询，查询1 到 r的“最近数字距离和”
// 找出中位数，存在mid中，然后通过简单计算即可，具体公式见代码
// 100 005 个数， tju 0.48，如果不超 int，不用long long
// 题目从0计，代码从 1 计，因此要++
#include <cstdio>
#include <string.h>
#include <algorithm>
using namespace std;
typedef long long LL;
#define lc(x)(x<<1)
#define rc(x)((x<<1) + 1)
const int M = 100005;
const int N = 18;       // 层数
struct node {
    int left, right;
    int mid() {
        return (left + right) >> 1;
    }
}tree[M*3];
int sorted[M];
int toleft[N][M];
int val[N][M];
LL sum[N][M];
LL s[M];
int cas = 1;
LL res, st;
void build(int l, int r, int level, int id);
int query(int l, int r, int k, int level, int id);

int main() {
    int n, m, l, r, k, mid;
    scanf("%d", &t);
    while(t--) {
        scanf("%d", &n);
        memset(sum, 0, sizeof(sum));
        s[0] = 0;
        for(int i = 1;i <= n;i++) {
            scanf("%d", &val[0][i]);
            sorted[i] = val[0][i];
            s[i] = s[i-1] + val[0][i];
        }
        sort(sorted + 1, sorted + n + 1);
        build(1, n, 0, 1);
        printf("Case #%d:\n", cas++);
        scanf("%d", &m);
        while(m--) {
            scanf("%d%d", &l, &r);
            l++;r++;
            st = 0;
            mid = query(1, n, (r - l + 1) >> 1, 0, 1);
            LL r1, r2, r3;
            r1 = (r-l+1)/2*(LL)mid;
            r1 -= st;
            r2 = s[r] - s[l-1] - (LL)mid - st;
            r3 = ((r-l)-(r-l+1)/2)*(LL)mid;
            res = r1 + r2 - r3;
            printf("%lld\n", res);
        }
        printf("\n");
    }
    return 0;
}

void build(int l, int r, int level, int id) {
    int lpos, rpos, same;
    tree[id].left = l;
    tree[id].right = r;
    if(tree[id].left == tree[id].right) return;
    int mid = tree[id].mid();
    int lsame = mid - l + 1;
    for(int i = l;i <= r;i++) {
        if(val[level][i] < sorted[mid])
            lsame--;
    }
    lpos = l;
    rpos = mid + 1;
    same = 0;
    for(int i = l;i <= r;i++) {
        if(i==1) {
            toleft[level][i] = 0;
        } else {
            toleft[level][i] = toleft[level][i-1];
        }
        if(val[level][i] < sorted[mid]) {
            toleft[level][i]++;
            val[level+1][lpos++] = val[level][i];
            sum[level][i] = sum[level][i-1] + (LL)val[level][i];
        } else if(val[level][i] > sorted[mid]) {
            val[level+1][rpos++] = val[level][i];
            sum[level][i] = sum[level][i-1];
        } else {
            if(same < lsame) {
                same++;
                toleft[level][i]++;
                val[level+1][lpos++] = val[level][i];
                sum[level][i] = sum[level][i-1] + (LL)val[level][i];
            } else {
                val[level+1][rpos++] = val[level][i];
                sum[level][i] = sum[level][i-1];
            }
        }
    }
    build(l, mid, level+1, lc(id));
    build(mid+1, r, level+1, rc(id));
    return;
}

int query(int l, int r, int k, int level, int id) {
    if (l==r) return val[level][l];
    int sl1, sl2, sr1, sr2, newl, newr, mid;
    if (l==tree[id].left) {
        sl1 = toleft[level][r];
        sl2 = 0;
    } else {
        sl1 = toleft[level][r] - toleft[level][l-1];
        sl2 = toleft[level][l-1];
    }
    // find in the left
    if (sl1 >= k) {
        newl = tree[id].left + sl2;
        newr = tree[id].left + sl2 + sl1 - 1;
        return query(newl, newr, k, level+1, lc(id));
    } else {
        mid = tree[id].mid();
        sr2 = l - tree[id].left - sl2;
        sr1 = r - l + 1 - sl1;
        newl = mid + sr2 + 1;
        newr = mid + sr2 + sr1;
        st += sum[level][r] - sum[level][l-1];
        return query(newl, newr, k-sl1, level+1, rc(id));
    }
}
```

```cpp
            val[level+1][lpos++] = val[level][i];
        } else if(val[level][i] > sorted[mid]) {
            val[level+1][rpos++] = val[level][i];
        } else {
            if(same < lsame) {
                same++;
                toleft[level][i]++;
                val[level+1][lpos++] = val[level][i];
            } else {
                val[level+1][rpos++] = val[level][i];
            }
        }
    }
    build(l, mid, level+1, lc(id));
    build(mid+1, r, level+1, rc(id));
    return;
}

int query(int l, int r, int k, int level, int id) {
    if (l==r) return val[level][l];
    int sl1, sl2, sr1, sr2, newl, newr, mid;
    if (l==tree[id].left) {
        sl1 = toleft[level][r];
        sl2 = 0;
    } else {
        sl1 = toleft[level][r] - toleft[level][l-1];
        sl2 = toleft[level][l-1];
    }
    // find in the left
    if (sl1 >= k) {
        newl = tree[id].left + sl2;
        newr = tree[id].left + sl2 + sl1 - 1;
        return query(newl, newr, k, level+1, lc(id));
    } else {
        mid = tree[id].mid();
        sr2 = l - tree[id].left - sl2;
        sr1 = r - l + 1 - sl1;
        newl = mid + sr2 + 1;
        newr = mid + sr2 + sr1;
        return query(newl, newr, k-sl1, level+1, rc(id));
    }
}
```

```cpp
    int size, val, minv, maxv, add, rev;
    int sum, lval, rval, cval;

    node() {
        size = 1;
        sum = 0;
        add = rev = 0;
        lc = rc = fa = NULL;

    void update() {
        int llv, lcv, lrv, ls, rlv, rcv, rrv, rs;
        size = 1;
        if(lc)size += lc -> size;
        if(rc)size += rc -> size;
        sum = minv = maxv = val;
        llv = lcv = lrv = rlv = rcv = rrv  = -INF;
        ls = rs = 0;
        if(lc) {
            minv = min(minv, lc -> minv);
            maxv = max(maxv, lc -> maxv);
            sum += lc -> sum;
            llv = lc -> lval;
            lcv = lc -> cval;
            lrv = lc -> rval;
            ls = lc -> sum;

            maxv = max(maxv, rc -> maxv);

            rlv = rc -> lval;
            rcv = rc -> cval;
            rrv = rc -> rval;
            rs = rc -> sum;

        rval = max(rrv, rs + val + max(0, lrv));
        cval = max(lcv, rcv);
        cval = max(cval, max(lrv + rlv, 0) + val);
        cval = max(cval, max(lrv, rlv) + val);
        return;
    }

            if (lc) {
                lc -> rev ^= 1;
                p = lc -> lc;
                lc -> lc = lc -> rc;
                lc -> rc = p;

                rc -> lc = rc -> rc;
```
