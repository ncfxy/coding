```cpp
            e[len] = 0;

                e[len]++;
                tm /= pri[i];
                w[len] *= pri[i];
            }
            len++;
        }
    }
    if (tm > 1) {
        p[len] = tm;
        w[len] = tm;
        e[len++] = 1;
    }

void sol() {
    LL res;
    if (m == 1) {
        printf("Case %d: 0\n", cas++);
        return;
    }
    if (k == n) {
        printf("Case %d: 1\n", cas++);
        return;
    }

        printf("Case %d: 0\n", cas++);
        return;

    res *= cal_c();

    int lofseg, rem;
    num[0] = 1;
    num[1] = 0;
    lofseg = 0;
    for (int i = 2;;i++) {
        num[i] = (LL)(i-1)*(num[i-1] + num[i-2]) % m;
        if (num[i] == 0 && i % m == 1 && num[i-1] == 1) {
            lofseg = i - 1; // 经过lofseg正好又回到初始状态
            break;
        }
    }
    rem = (n - k) % lofseg;
    return num[rem];
}

int cal_c() {
    int ans;
    for (int i = 0;i < len;i++) {
        pi = p[i];
        pi_e = w[i];
        cnt_of_p = 0;
        get_f_inv();
        b[i] = cal_f(n);
        b[i] *= cal_finv(k);b[i] %= pi_e;
        b[i] *= cal_finv(n-k);b[i] %= pi_e;
        b[i] *= mul_mod(pi, cnt_of_p, pi_e); b[i] %= pi_e;
    }
    // x == b[i] mod(w[i]) (o <=i < len)
    ans = CRT();
    return ans;
}

void get_f_inv() {
    f[0] = finv[0] = f[1] = finv[1] = 1;
    cnt_of_p = 0;
    for (int i = 2;i <= pi_e; i++) {
        f[i] = f[i-1]; finv[i] = finv[i-1];     // p的倍数的f值和inv值都没有，因为后面有运算*k!*p^k
        if(i % pi) {
            f[i] = f[i] * i % m;
            finv[i] = finv[i] *cal_inv(i) % m;
        }
    }
    return;
}

int cal_inv(int k) {
    int x, y;
    ext_gcd(k, pi_e, x, y);
    while(x < 0) x += pi_e;
    return x;
}

void ext_gcd(int aa, int bb, int &x, int &y) {
    if(bb == 0) {
        x = 1;
        y = 0;
        return;
    }
    int tx;
    ext_gcd(bb, aa % bb, x, y);
    tx = x;
    x = y;
    y = tx - aa / bb * y;
    return;
}

int cal_f(int n) {
    if(n <= 1) return 1;
    if cnt_of_seg = n/pi_e, rem = n % pi_e, ans;
    ans = mul_mod(f[pi_e - 1], cnt_of_seg, pi_e);
    ans *= f[rem];
    ans %= pi_e;
    n /= pi;
    cnt_of_p += n;
    ans *= cal_f(n);
    ans %= pi_e;
    return ans;
}

int cal_finv(int n) {
    if(n <= 1) return 1;
    LL cnt_of_set = n / pi_e, rem = n % pi_e, ans;
    ans = mul_mod(finv[pi_e - 1], cnt_of_set, pi_e);
    ans *= finv[rem];
    ans %= pi_e;
    n /= pi;
    cnt_of_p -= n;
    ans *= cal_finv(n);
    ans %= pi_e;
    return ans;
}

int mul_mod(LL a, int k, int p) {
    LL ans = 1, tmp = a % p;
    while(k) {
        if(k & 1) ans = (ans * tmp) % p;
        tmp = (tmp * tmp) % p;
        k >>= 1;
    }
    return ans;
}

int CRT() {
    int x, y, ans = 0, m,n = 1;
    for (int i = 0;i < len;i++) {
        n *= w[i];
    }
    for (int i = 0;i < len;i++) {
        m = n / w[i];
        ext_gcd(w[i], m, x, y);
        LL ty = y;
        ty = ty * m % n;
        ty = ty * b[i] % n;
        ans = (ans + ty) % n;
    }
    if(ans >= 0) return ans;
    return ans + n;
}
```

## 2.13 中国剩余定理（模数 mi 不互素）
```cpp
// 3680 T == [1,100] N == [1,6] Mi==[1,50] Ai=[0,Mi) 0.00s
// x = ai(mod bi)
// 求是否存在解，如果存在，输出最小值（先输入mi，再输入余数ai）

// 分析
// x = qi*bi + ai = qi*ki*d + ai
// x = qj*bj + aj = qj*kj*d + aj

// 其中n[i] = ai%p[i]^e[i]
// 对于不同bi, bj 有相同 p[i]，留次数最高的一个
// 最后用中国剩余定义求出答案，即为最小值
// 注意，模板不要求一定能打出素数表，可以有if(temp > 1) 的情况

#include <cstdio>
#include <string.h>
#include <string>
typedef long long LL;

// mi 能够得到的最大的素数个数，如题目bi <= 50，则MAXM >= sqrt(50) 即可
const int MAXM = 8;
// 原方程组的个数
const int MAXN = 7;
int pri[MAXM];
int cover[MAXM];
int top;

void prim(){
    memset(cover, 0, sizeof(cover));
    top = 0;
    for(int i = 2;i < MAXM;i++){
        if(cover[i] == 0){
            pri[top++] = i;
        }
        for(int j = 0;j < top && pri[j] <= MAXM / i;j++){
            cover[i * pri[j]] = 1;
            if(i % pri[j] == 0){
                break;
            }
        }
    }
    return;
}
int gcd(int a, int b){
    return (b == 0 ? a : gcd(b, a % b));
}

void init();
void split(int id);
void ex_gcd(int a, int b, int &x, int &y);
void sol();
LL CRT();

int a[MAXN], b[MAXN];
// fac[i][j][0] means mi's jth prim number
// fac[i][j][1] means number of mi's jth prim numbers
// fac[i][j][2] means protect of mi's jth prim number
int fac[MAXN][MAXM][3];
int rem[MAXN];
int nm[MAXN];
int cov[MAXN];
int numf;
int flag;
// 以下变量为 CRT 所需要的
int n[MAXM + MAXN], m[MAXM + MAXN];
int num;
int cas = 1;
int main() {
    int t;
    scanf("%d", &t);
    prim();
    while(t--) {
        init();
        sol();
    }
    return 0;
}

void init() {
    int d;
    memset(cov, 0, sizeof(cov));
    memset(nm, 0, sizeof(nm));
    memset(rem, -1, sizeof(rem));
    num = 1;
    flag = 0;
    scanf("%d", &numf);
    for(int i = 1;i <= numf;i++) {
        scanf("%d", &b[i]);
    }
    for(int i = 1;i <= numf;i++) {
        scanf("%d", &a[i]);
    }
    for(int i = 1; i <= numf;i++) {
        for (int j = i + 1;j <= numf;j++) {
            d = a[i] - a[j];
            if (d < 0) d = -d;
            if (d % gcd(b[i], b[j]) != 0) {
                flag = 1;
                return;
            }
        }
    }
    for(int i = 1;i <= numf;i++) {
        split(i);
    }
    for(int i = 1;i <= num;i++) {
        for(int j = 0; j < nm[i];j++) {
            if(cov[fac[i][j][0]] == fac[i][j][1]) {
                cov[fac[i][j][0]] = -1;
                n[num] = a[i] % fac[i][j][2];
                m[num++] = fac[i][j][2];
            }
        }
    }

    // 处理 rem
    for(int i = 1; i <= numf;i++) {
        if(rem[i] == -1) continue;
        for(int j = i + 1; j <= numf;j++) {
            if(rem[i] == rem[j]) {
                rem[j] = -1;
            }
        }
    }
    for(int i = 1;i <= numf;i++) {
        if(rem[i] != -1) {
            n[num] = a[i] % rem[i];
            m[num++] = rem[i];
        }
    }
    num--;
    return;
}
void split(int id) {
    int temp = b[id], tmp;
    for(int i = 0;i < top && pri[i] <= temp;i++) {
        if(temp % pri[i] == 0) {
            // p
            fac[id][nm[id]][0] = i;
            // e
            fac[id][nm[id]][1] = 0;
            tmp = temp;
            while(temp % pri[i] == 0) {
                fac[id][nm[id]][1]++;
                temp /= pri[i];
            }
            // p ^ e
            fac[id][nm[id]][2] = tmp / temp;
            if(cov[i] < fac[id][nm[id]][1]) {
                cov[i] = fac[id][nm[id]][1];
            }
            nm[id]++;
        }
    }
    if(temp > 1) {
        rem[id] = temp;
    }
    return;
}

void sol() {
    if(flag) {
        printf("Case %d: -1\n", cas++);
        return;
    }
    printf("Case %d: %lld\n", cas++, CRT());
    return;
}
void ex_gcd(int a, int b, int &x, int &y) {
    if(b == 0) {
        x = 1;
        y = 0;
        return;
    }
    LL tx;
    ex_gcd(b, a % b, x, y);
    tx = x;
    x = y;
    y = tx - (a / b) * y;
    return;
}

LL CRT() {
    LL ans, mm, M, x, y;
    ans = 0;
    M = 1;
    for(int i = 1;i <= num;i++) M *= m[i];
    for(int i = 1;i <= num;i++) {
        mm = M / m[i];
        ex_gcd(mm%m[i], m[i], x, y);
        ans += (n[i] * mm * (x % m[i]));
        ans %= M;
    }
    if (ans <= 0) ans += M;
    return ans;
}
```

## 实数高斯消元
```cpp
// 实数高斯消元 O(n^3)
// 输入 m 行 n + 1 列数组，分别表示 m 个方程 n 个未知数
// 返回 0 表示无解， 1 表示唯一解， -1 表示无穷解
// res[] 给出一组可行解，自由变量 tag[] 里标出
// TOI 1111
// 注意是 m 行 n+1 列
#include <cstdio>
#include <string.h>
#include <cmath>
#include <algorithm>

using namespace std;
typedef double db;

const int M = 105;
const db EPS = 1e-8;
const db NZ = 0.005;    // 防止 -0
inline int sig(db k) {if(k < -EPS) return -1; return k > EPS;}
int n, m;
db a[M][M]; // 系数矩阵
db res[M];  // 一组可行解
int tag[M]; // 自由变量
void init();
void sol();
int gauss();

int main() {
    // 注意是 m 行 n+1 列
    // freopen("gaess.in", "r", stdin);
    // freopen("gaess.out", "w", stdout);
    while(scanf("%d%d", &n, &m) != EOF) {
        init();
        sol();
    }
    return 0;
}

void init() {
    n = m;
    for(int i = 1;i <= m;i++) {
        for(int j = 1;j <= n + 1;j++) scanf("%lf", &a[i][j]);
    }
    return;
}
void sol() {
    int flag;
    flag = gauss();
    if(flag == 0) {
        printf("No Solution\n");
        return;
    }
    if(flag == -1) {
        printf("Multiple solution\n");
        return;
    }
    // else
        // printf("Single solution\n");
    for(int i = 1;i <= n;i++) {
        if(fabs(res[i]) < NZ) res[i] = 0.0;
    }
    for(int i = 1;i <= n;i++) {
        printf("x%d=%.2lf\n", i, res[i]);
    }
    return;
}

int gauss() {
    int nr, nc, maxi, multi;
    db temp;
    memset(tag, 0, sizeof(tag));
    memset(res, 0, sizeof(res));
    multi = 0;
    for(nr = nc = 1; nr <= m && nc <= n;) {
        temp = 0.0;
        for(int i = nr; i <= m;i++) {
            if(fabs(a[i][nc]) > temp) {
                temp = fabs(a[i][nc]);
                maxi = i;
            }
        }
        if(sig(temp) == 0) {
            tag[nc++] = 1;
            multi = 1; 
            continue;
        }
        for(int j = nc;j <= n+1; j++) {
            swap(a[maxi][j], a[nr][j]);
        }
        for(int j = n + 1;j >= nc; j--) {
            a[nr][j] /= a[nr][nc];
        }
        for(int i = 1;i <= m;i++) {
            for(int j = n + 1;j >= nc;j--) {
                if(i != nr) {
                    a[i][j] -= a[nr][j] * a[i][nc];
                }
            }
        }
        nr++;
        nc++;
    }
    for(int i = nr;j <= m;i++) {
        if(sig(a[i][n+1]) != 0) return 0;
    }
    if(nr < n) multi = 1;
    for(int i = 1;i <= m;i++) {
        int j;
        for(j = 1;j <= n;j++) {
            if(sig(a[i][j]) != 0) break;
        }
        if(j <= n) res[j] = a[i][n+1];
    }
    return multi ? -1 : 1;
}
```

## 2.15 整数高斯消元
```cpp
// 整数高斯消元 O(n^3)
// m 行 n+1 列数组，m 个方程 n 个未知数
// note: 乘法运算很容易溢出，实际应用常用大数运算
// hdu 2449 java
#include <cstdio>
#include <string.h>
#include <cmath>
#include <algorithm>
using namespace std;

typedef long long LL;
typedef double db;
const int M = 105;
const db NZ = 0.005;

LL a[M][M];
db res[M];
int n, m;

inline LL gcd(LL a, LL b) {
    if(a < 0) a = -a;
    if(b < 0) b = -b;
    return b == 0 ? a : gcd(b, a % b);
}

void init();
void sol();
void gauss_int();

int main() {
    // m 行 n+1 列方程组
    while(scanf("%d", &m) != EOF) {
        init();
        sol();
    }
    return 0;
}

void init() {
    n = m;
    for(int i = 1;i <= m;i++) {
        for (int j = 1;j <= n + 1;j++) scanf("%I64d", &a[i][j]);
    }
    return;
}

void sol() {
    int flag, j;
    gauss_int();
    flag = 1;
    for(int i = 1;i <= m;i++) {
        int j;
        for(j = 1;j <= n;j++) {
            if(a[i][j]) break;
        }
        if (j == n+1) {
            if(a[i][n+1]) {
                flag = 0;
                break;
            } else {
                flag = -1;
                break;
            }
        }
        res[i] = (db)a[i][n+1]/a[i][j];
    }
    if(flag != 1) {
        printf("%d\n", flag);
    } else {
        for(int i = 1;i <= n;i++) {
            if(fabs(res[i]) < NZ) res[i] = 0.0;
        }
        for(int i = 1;i <= n;i++) {
            printf("x%d=%.2lf\n", i, res[i]);
        }
    }
    return;
}

void gauss_int() {
    int row, col, tr, tc, fst;
    LL d, temp, gd;
    for(row = col = 1; col <= n; col++) {
        for(tr = row; tr <= m; tr++) { // 用 tr 行 col列的系数消去其他行 col 列的系数
            if(a[tr][col]) break;
        }
        if(tr == m+1) continue;         // 无穷多组解
        if(tr != row) {
            for(tc = 1; tc <= n+1; tc++) {
                swap(a[tr][tc], a[row][tc]);
            }
        }
        for(tr = 1; tr <= m;tr++) {
            if(tr == row) continue;
            temp = a[tr][col];
            fst = 1;
            for(tc = 1; tc <= n+1; tc++) {
                a[tr][tc] = a[tr][tc] * a[row][col] - a[row][tc] * temp;
                if(a[tr][tc] != 0) {
                    if(fst == 1) {
                        gd = a[tr][tc];
                        fst = 0;
                    } else {
                        gd = gcd(gd, a[tr][tc]);
                    }
                }
            }
        }
        row++;
    }
    return;
}
```

```java
// 整数高斯消元（化成分数）

import java.util.*;
import java.math.*;

public class Main{
    public static void main(String[] args){
        Scanner io = new Scanner(System.in);
        int M = 105;

        BigInteger [][]a = new BigInteger[M][M];
        BigInteger []res1 = new BigInteger[M];
        BigInteger []res2 = new BigInteger[M];

            m = io.nextInt();
            n = m;
            int i,j;
            // init()
            for(i = 1;i <= m;i++){
                for (j = 1;j <= n+1;j++) a[i][j] = io.nextBigInteger();
            }
            int flag;
            // gauss()
            int row, col, tr, tc, fst;
            gd = BigInteger.ONE;
            flag = 1;
            for(row = col = 1; col <= n; col++){
                for(tr = row; tr <= m; tr++){
                    if(!a[tr][col].equals(BigInteger.ZERO)) break;
                }
                if (tr == m+1) continue;
                if(tr != row){
                    for(tc = 1; tc <= n+1; tc++){
                        temp = a[tr][tc];
                        a[tr][tc] = a[row][tc];
                        a[row][tc] = temp;
                    }
                }
                for(tr = 1; tr <= m;tr++){
                    if(tr == row) continue;
                    temp = a[tr][col];
                    fst = 1;
                    for(tc = 1; tc <= n+1; tc++){
                        temp1 = a[tr][tc].multiply(a[row][col]);
                        temp2 = a[row][tc].multiply(temp);
                        a[tr][tc] = temp1.subtract(temp2);
                        if(!a[tr][tc].equals(BigInteger.ZERO)){
                            if(fst == 1){
                                gd = a[tr][tc];
                                fst = 0;
                            } else {
                                gd = gd.gcd(a[tr][tc]);
                            }
                        }
                    }
                    if(fst == 0){
                        for(tc = 1; tc <= n+1; tc++){
                            a[tr][tc] = a[tr][tc].divide(gd);
                        }
                    }
                }
                row++;
            }
            // sol()
            for(i = 1;i <= m;i++){
                for(j = 1;j <= n;j++){
                    if(!a[i][j].equals(BigInteger.ZERO)) break;
                }
                if(j == n+1){
                    if(!a[i][n+1].equals(BigInteger.ZERO)){
                        flag = 0;
                        break;

                        flag = -1;
                        break;
                    }

                res1[i] = a[i][n+1];

            else {
```
