## 5.1 高精度
```cpp
#include <cstdio>
#include <string.h>
#include <string>

const int LENGTH = 200; // 能保存的数字长度为LENGTH * 4

struct BigNum {
  bool sign;
  int num[LENGTH];
  int len;
  BigNum() {
    sign = true;
    memset(num, 0, sizeof(num));
    len = 1;
  }

  BigNum(int n) {
    int b;
    if (n >= 0) {
      sign = true;
      b = n;
    } else {
      sign = false;
      b = -n;
    }
    memset(num, 0, sizeof(num));
    len = 0;
    while (b) {
      num[len++] = b % 10000;
      b /= 10000;
    }
    if (!len)
      len = 1;
  }
};

// 比较: 相等返回0， a > b 返回正数，否则返回负数
int compare(const BigNum &a, const BigNum &b);
// 输入，从char[] str里读到BigNum 里
void input(BigNum &a, const char str[]);
// 输出
void print(const BigNum &a);
// 加法 c = a + b, 用到compare, subtract
void plus(const BigNum &a, const BigNum &b, BigNum &c);
// 减法 c = a - b, 用到compare, plus
void subtract(const BigNum &a, const BigNum &b, BigNum &c);
// 加法 a += b
void increase(BigNum &a, const BigNum &b);
// 减法 a -= b
void decrease(BigNum &a, const BigNum &b);
// 乘法 c = a * b, 用到compare, plus, subtract
void multiply(const BigNum &a, const BigNum &b, BigNum &c);
// 乘10，处罚所需
void multiply10(BigNum &a);
// 除法 c = a / b, d为余数，用到compare, multiply10, subtract, plus
void divide(const BigNum &a, const BigNum &b, BigNum &c, BigNum &d);

void input(BigNum &a, const char str[]) {
  int i, j, len;
  i = 0;
  memset(a.num, 0, sizeof(a.num));
  if (str[i] == '-') {
    i++;
    a.sign = false;
  } else {
    a.sign = true;
  }
  while (str[i] == '0' && (int)strlen(str) != i + 1)
    i++;
  len = strlen(str) - 1;
  a.len = 0;
  while (len >= i) {
    for (j = 1; j < 10000 && len >= i; j *= 10)
      a.num[a.len] += j * (str[len--] - '0');
    a.len++;
  }
  if (a.len == 1 && a.num[0] == 0)
    a.sign = true;
  return;
}

void print(const BigNum &a) {
  int i;
  if (!a.sign)
    printf("-");
  // else printf("+");
  printf("%d", a.num[a.len - 1]);
  for (i = a.len - 2; i >= 0; i--)
    printf("%04d", a.num[i]);
  printf("\n");
  return;
}
// 比较: 相等返回0， a > b 返回正数，否则返回负数
int compare(const BigNum &a, const BigNum &b) {
  if (a.sign != b.sign)
    return (a.sign ? 1 : -1);
  if (a.len > b.len)
    return a.sign ? 1 : -1;
  else if (a.len < b.len)
    return a.sign ? -1 : 1;
  int len = a.len - 1;
  while (len >= 0 && a.num[len] == b.num[len])
    len--;
  return (a.sign ? 1 : -1) * (a.num[len] - b.num[len]);
}

// 加法 c = a + b, 用到compare, subtract
void plus(const BigNum &a, const BigNum &b, BigNum &c) {
  if (a.sign != b.sign) {
    // 调用高精度减法
    BigNum d = b;
    d.sign = !b.sign;
    subtract(a, d, c);
    return;
  }
  memset(c.num, 0, sizeof(c.num));
  c.sign = a.sign;
  c.len = a.len > b.len ? a.len : b.len;
  int i, ca = 0;
  for (i = 0; i < c.len; i++) {
    c.num[i] += a.num[i] + b.num[i] + ca;
    ca = c.num[i] / 10000;
    c.num[i] %= 10000;
  }
  if (ca)
    c.num[c.len++] = ca;
  if (c.len == 1 && c.num[0] == 0)
    c.sign = true;
  return;
}

// 加法 a += b
void increase(BigNum &a, const BigNum &b){
    if(a.sign != b.sign) {
        BigNum d = b;
        d.sign = !b.sign;
        decrease(a,d);
        return;
    }
    int i, ca = 0;
    if(a.len < b.len) a.len = b.len;
    for(i = 0;i < a.len;i++) {
        a.num[i] += b.num[i] + ca;
        ca = a.num[i] / 10000;
        a.num[i] %= 10000;
    }
    if(ca) a.num[a.len++] = ca;
    if(a.len == 1 && a.num[0] == 0) a.sign = true;
    return;
}
// 减法 c = a - b, 用到compare, plus
void subtract(const BigNum &a, const BigNum &b, BigNum &c){
    if(a.sign != b.sign) {
        // 调用高精度加法
        BigNum d = b;
        d.sign = !b.sign;
        plus(a,d,c);
        return;
    }
    if(!a.sign) {
        BigNum e = a, d = b;
        e.sign = d.sign = true;
        subtract(d, e, c);
        return;
    }
    memset(c.num, 0, sizeof(c.num));
    int i, ca, ret = compare(a,b);
    if (ret == 0) {
        c = 0;
        return;
    } else if(ret > 0) {
        c.sign = true;
        ca = 0;
        for(i = 0;i < a.len; i++) {
            c.num[i] = a.num[i] - b.num[i] - ca;
            if(c.num[i] <0) {
                c.num[i] += 10000;
                ca = 1;
            }else {
                ca = 0;
            }
        }
        c.len = a.len;
        while(c.len > 1 && c.num[c.len-1] == 0) c.len--;
    } else {
        c.sign = false; ca = 0;
        for (i = 0;i < b.len; i++) {
            c.num[i] = b.num[i] - a.num[i] - ca;
            if(c.num[i] < 0) {
                c.num[i] += 10000;
                ca = 1;
            }else {
                ca = 0;
            }
        }
        c.len = b.len;
        while(c.len > 1 && c.num[c.len-1] == 0) c.len--;
    }
    if(c.len == 1 && c.num[0] == 0) c.sign = true;
    return;
}
// 减法 a -= b
void decrease(BigNum &a, const BigNum &b){
    if (a.sign != b.sign) {
        // 调用加法
        BigNum d = b;
        d.sign = !b.sign;
        increase(a, d);
        return;
    }
    if (!a.sign) {
        BigNum d = b;
        a.sign = d.sign = true;
        decrease(a,d);
        a.sign = !a.sign;
        return;
    }
    int i, ca, ret = compare(a, b);
    if (ret == 0) {
        a = 0;
        return;
    }else if (ret > 0) {
        ca = 0;
        for(i = 0;i < a.len;i++){
            a.num[i] = a.num[i] - b.num[i] - ca;
            if(a.num[i] < 0) {
                a.num[i] += 10000;
                ca = 1;
            }else {
                ca = 0;
            }
        }
        while(a.len > 1 && a.num[a.len-1] == 0) a.len--;
    }else {
        a.sign = false; 
        ca = 0;
        for(i = 0;i < b.len;i++) {
            a.num[i] = b.num[i] - a.num[i] - ca;
            if(a.num[i] < 0) {
                a.num[i] += 10000;
                ca = 1;
            }else {
                ca = 0;
            }
        }
        while(a.len > 1 && a.num[a.len-1] == 0) a.len--;
    }
    if(a.len == 1 && a.num[0] == 0) a.sign = true;
    return;
}

// 乘法 c = a * b, 用到compare, plus, subtract
void multiply(const BigNum &a, const BigNum &b, BigNum &c){
    c.sign = (a.sign == b.sign ? true: false);
    memset(c.num, 0, sizeof(c.num));
    int i, j;
    for(i = 0;i < a.len;i++) {
        for(j = 0;j < b.len;j++) {
            c.num[i+j] += (a.num[i] * b.num[j]);
            c.num[i+j+1] += c.num[i+j] / 10000;
            c.num[i+j] %= 10000;
        }
    }
    c.len = a.len + b.len;
    while(c.len > 1 && c.num[c.len-1] == 0) c.len--;
    if(c.len == 1 && c.num[0] == 0) c.sign = true;
    return;
}
// 乘10，除法所需
void multiply10(BigNum &a){
    int len, i, ca;
    len = a.len;
    a.num[len - 1] *= 10;
    ca = a.num[len -1] / 10000;
    if(ca) {
        a.num[len] = ca;
        a.len++;
    }
    for(i = len - 2; i >= 0; i--) {
        a.num[i] *= 10;
        a.num[i+1] += a.num[i] / 10000;
        a.num[i] %= 10000;
    }
    return;
}

// 除法 c = a / b, d为余数，用到compare, multiply10, subtract, plus
void divide(const BigNum &a, const BigNum &b, BigNum &c, BigNum &d){
    c.sign = (a.sign == b.sign ? true: false);
    d.sign = true;
    memset(c.num, 0, sizeof(c.num));
    memset(d.num, 0, sizeof(d.num));
    if(!compare(b, 0)) return;
    BigNum e, h;
    int i, j, p;
    c.len = a.len;
    d.len = 1;
    h = b;
    h.sign = true;
    for(i = c.len -1; i>=0;i--) {
        p = a.num[i];
        for(j = 1000; j >= 1; j/=10) {
            multiply10(d);
            d.num[0] += (p/j);
            p %= j;
            while(compare(d, h) >= 0) {
                subtract(d,h,e);
                d = e;
                c.num[i] += j;
            }
        }
    }
    while(c.len > 1 && c.num[c.len - 1] == 0) c.len--;
    if (compare(d,0) && !a.sign) {
        j = c.sign? 1: -1;
        plus(c, j, e);
        c = e;
        subtract(h, d, e);
        d = e;
    }
    if (c.len == 1 && c.num[0] == 0) c.sign = true;
    return;
}
```

```cpp
  if (a.len > b.len)
    return 1;
  else if (a.len < b.len)
    return -1;

// 加法 a += b
void increase(BigNum &a, const BigNum &b) {

    a.len = b.len;

void subtract(const BigNum &a, const BigNum &b, BigNum &c) {

  if (ret <= 0) {
    c = 0;
    return;

  for(i = 0;i < a.len; i++) {
    c.num[i] = a.num[i] - b.num[i] - ca;

        ca = 0;
    }

  c.len = a.len;
  while(c.len > 1 && c.num[c.len-1] == 0) c.len--;
  return;
}
```
