```cpp
#include <cstdio>
#include <iostream>

using namespace std;
const double EPS = 1e-8;
const int M = 1005;

struct node {
  double x;
  double y;
} p[M];
// 判断点是否在直线上
bool PoolLine(const node &a, const node &b, const node &c);
// 判断线段是否相交
bool IsIntersected(const node &a, const node &b, const node &c, const node &d);

double Max(double a, double b);
double Min(double a, double b);

// 判断 c 是否在a，b某一侧
double Direction(const node &a, const node &b, const node &c);
// 求交点 Xc 坐标
double CountXc(const node &a, const node &b, const node &c, const node &d);
// 求交点 Yc 坐标
double CountYc(const node &a, const node &b, const node &c, const node &d);
// 求凸多边形面积
double CountArea(const node seq[], int n);
// 矩阵乘法(2*2)
double Matrix(const node &a, const node &b);

bool PoolLine(const node &a, const node &b, const node &c) {
  double r;
  r = Direction(a, b, c);
  if (Max(a.x, b.x) >= c.x && Min(a.x, b.x) <= c.x && Max(a.y, b.y) >= c.y &&
      Min(a.y, b.y) <= c.y && r > -EPS && r < EPS)
    return true;
  else
    return false;
}

bool IsIntersected(const node &a, const node &b, const node &c, const node &d) {
  if (Max(a.x, b.x) >= Min(c.x, d.x) && Min(a.x, b.x) <= Max(c.x, d.x) &&
      Max(a.y, b.y) >= Min(c.y, d.y) && Min(a.y, b.y) <= Max(c.y, d.y) &&
      Direction(a, b, c) * Direction(a, b, d) < -EPS &&
      Direction(c, d, a) * Direction(c, d, b) < -EPS) {
    return true;
  } else {
    return false;
  }
}

double Max(double a, double b) {
    if(a  - b > EPS) return a;
    else return b;
}

double Min(double a, double b) {
    if(a - b < -EPS) return a;
    else return b;
}

double Direction(const node &a, const node &b, const node &c) {
    return (a.x - b.x) * (a.y - c.y) - (a.y - b.y) * (a.x - c.x);
}

double CountXc(const node &a, const node &b, const node &c, const node &d) {
    double x, Area1, Area2;
    node seq1[3], seq2[3];
    seq1[0] = a; seq1[1] = c; seq1[2] = b;
    seq2[0] = a; seq2[1] = d; seq2[2] = b;
    Area1 = CountArea(seq1, 3);
    Area2 = CountArea(seq2, 3);
    // 面积乘以对角坐标
    x = (Area1 * d.x + Area2 * c.x) / (Area1 + Area2);
    return x;
}

double CountYc(const node &a, const node &b, const node &c, const node &d) {
    double y, Area1, Area2;
    node seq1[3], seq2[3];
    seq1[0] = a; seq1[1] = c; seq1[2] = b;
    seq2[0] = a; seq2[1] = d; seq2[2] = b;
    Area1 = CountArea(seq1, 3);
    Area2 = CountArea(seq2, 3);
    // 面积乘以对角坐标
    y = (Area1 * d.y + Area2 * c.y) / (Area1 + Area2);
    return y;
}

double CountArea(const node seq[], int n) {
    double r = 0;
    for(int i = 0;i < n;i++) {
        if(i < n-1) r += Matrix(seq[i], seq[i+1]);
        else r += Matrix(seq[i], seq[0]);
    }
    if (r < -EPS) r = -r;
    return r;
}

double Matrix(const node &a, const node &b) {
    double r;
    r = ((a.x * b.y) - (b.x * a.y)) / 2;
    return r;
}
```

## 4.2 求对称点
```cpp
#include <cstdio>
#include <string>

const double EPS = 1e-6;

double lx1, ly1;
double lx2, ly2;
double px1, py1;

int flag1;

double lk, lb;
double xt, yt;

void Find(double px0, double py0) {
  flag1 = 0;
  if (ly1 - ly2 >= -EPS && ly1 - ly2 <= EPS) {
    lk = 0;
    lb = ly1;
    xt = px1;
    yt = 2 * ly1 - py1;
    return;
  } else if (lx1 - lx2 >= -EPS && lx1 - lx2 <= EPS) {
    flag1 = -1;
    xt = 2 * lx1 - px1;
    yt = py1;
    return;
  }
  lk = (ly1 - ly2) / (lx1 - lx2);
  lb = ly1 - lk * lx1;
  xt = (2 * lk * py0 - (lk * lk - 1) * px0 - 2 * lk * lb) / (1 + lk * lk);
  yt = (2 * lk * px0 + (lk * lk - 1) * py0 + 2 * lk) / (1 + lk * lk);
  return;
}
``` 

```cpp
#include <stdio.h>

/// 向量叉积
inline double cross(const Vector &a, const Vector &b){
    return a.x * b.y - a.y * b.x;
}

inline double scalar(const Vector &a, const Vector &b){
    return a.x * b.x + a.y * b.y;
}

    return sqrt((a.x - b.x) * (a.x - b.x) + (a.y - b.y) * (a.y - b.y));
}

inline bool pointAboveSegment(const Point &p, const Point &a, const Point &b){
    Vector v1, v2;
    v1.x = p.x - a.x; v1.y = p.y - a.y;
    v2.x = b.x - a.x; v2.y = b.y - a.y;
    if(scalar(v1, v2) < 0) return false;
    v1.x = p.x - b.x; v1.y = p.y - b.y;
    v2.x = a.x - b.x; v2.y = a.y - b.y;
    return scalar(v1, v2) >= 0;
}

bool pointOnSegment(const Point &p, const Point &a, const Point &b){
    Vector v1, v2;
    v1.x = a.x - b.x; v1.y = a.y - b.y;
    v2.x = p.x - b.x; v2.y = p.y - b.y;
    if (fabs(cross(v1, v2)) > 1e-6) return false;
    if (p.x <= Max(a.x, b.x) && p.x >= Min(a.x, b.x) &&
        p.y <= Max(a.y, b.y) && p.y >= Min(a.y, b.y)) return true;
    else return false;

double pointToLine(const Point &p, const Point &a, const Point &b){
    Vector v1, v2;
    v1.x = a.x - p.x; v1.y = a.y - p.y;
    v2.x = b.x - p.x; v2. y = b.y - p.y;
    double area = fabs(cross(v1, v2));
    return area / lengthOfSegment(a,b);

int segmentIntersect(const Point &sa, const Point &sb, const Point &ta, const Point &tb){
    // 此处快速判断根据具体情况而来
    // 快速排斥
    if (Max(sa.x , sb.x) < Min(ta.x, tb.x)) return 0;
    if (Max(ta.x, tb.x) < Min(sa.x, sb.x)) return 0;
    if (Max(sa.y, sb.y) < Min(ta.y, tb.y)) return 0;
    if (Max(ta.y, tb.y) < Min(sa.y, sb.y)) return 0;
    // 跨立实验
    Vector v1, v2, v3; double pro;
    v1.x = tb.x - ta.x; v1.y = tb.y - ta.y;
    v2.x = sa.x - ta.x; v2.y = sa.y - ta.y;
    v3.x = sb.x - ta.x; v3.y = sb.y - ta.y;
    pro = cross(v2, v1) * cross(v1, v3);
    if (pro < -1e-6) return 0;
    else if(pro < 1e-6) {
        if (pointOnSegment(sa, ta, tb) || pointOnSegment(sb, ta, tb)) return -1;
    }
    v1.x = sb.x - sa.x; v1.y = sb.y - sa.y;
    v2.x = ta.x - sa.x; v2.y = ta.y - sa.y;
    v3.x = tb.x - sa.x; v3.y = tb.y - sa.y;
    pro = cross(v2, v1) * cross(v1, v3);
    if (pro < -1e-6) return 0;
    else if(pro < 1e-6) {
        if (pointOnSegment(ta, sa, sb) || pointOnSegment(tb, sa, sb)) return -1;
    }
    return 1;
}

// using segmentIntersect, pointOnSegment, cross
int pointInPolygon(const Point pol[], const int sd, const Point &a){
    int i, j;
    int segInte, count = 0;
    Point up;
    Point left = {MIN_D, a.y};
    for(int i = 0;i < sd; i++) {
        j = (i + 1) % sd;
        if(pointOnSegment(a, pol[i], pol[j])) return -1;
        if (pol[i].y != pol[j].y) {
            segInte = segmentIntersect(left, a, pol[i], pol[j]);
            up = pol[i].y > pol[j].y ? pol[i] : pol[j];
            if (segInte < 0 && pointOnSegment(up, left, a)) count++;
            else if(segInte > 0) count++;
        }
    }
    return count & 1;
}

// 线段是否在多边形内，如果线段在多边形上，返回-1，内外都有，返回2
// using pointInPolygon, segmentIntersect, pointOnSegment, cross, cmp
int SegmentInPolygon(const Point pol[], const int sd, const Point &a, const Point &b){
    int i, j;
    int on, in, ptIn;
    for(i = 0;i < sd;i++) {
        j = (i + 1) % sd;
        if (segmentIntersect(pol[i], pol[j], a, b) > 0) return 2;
    }
    Point midPt;
    Point dot[SIDE];
    on = 0;
    for(i = 0;i < sd;i++)
        if(pointOnSegment(pol[i], a, b)) dot[on++] = pol[i];
    std::sort(dot, dot + on, cmp);
    in = -1;
    for(i = 1;i < on;i++) {
        midPt.x = (dot[i-1].x + dot[i].x) / 2;
        midPt.y = (dot[i-1].y + dot[i].y) / 2;
        if(in != -1) {
            ptIn = pointInPolygon(pol, sd, midPt);
            if(ptIn != -1 && in != ptIn)return 2;
        } else {
            in = pointInPolygon(pol, sd, midPt);
        }
    }
    return in;
}

// 点到多边形的距离
// using pointInPolygon, segmentIntersect, pointOnSegment, cross, 
// pointAboveSegment, pointToLine, cross, lengthOfSegment, scalar
double pointToPolygon(const Point pol[], const int sd, const Point &a){
    if(pointInPolygon(pol, sd, a)) return 0.0;
    int i,j;
    double ptToPol = MAX_D;
    for(i = 0;i < sd;i++) {
        j = (i + 1) % sd;
        double temp = lengthOfSegment(a, pol[i]);
        if(ptToPol > temp) ptToPol = temp;
        if(pointAboveSegment(a, pol[j], pol[i])){
            double temp = pointToLine(a, pol[j], pol[i]);
            if(ptToPol > temp) ptToPol = temp;
        }
    }
    return ptToPol;
}
```

### 理论基础
1. 向量叉积
    \(x_1y_2 - y_1x_2\)

    \(|\vec{c}| = |\vec{a} \times \vec{b}| =  |\vec{a}||\vec{b}|\sin\theta\)
2. 向量内积
    \(x_1x_2 + y_1y_2\)

    \(\vec{a} \cdot \vec{b} = |\vec{a}| |\vec{b}| \cos\theta\)
3. 线段长度
    \(\sqrt{(x_1 - x_2)^2+(y_1 - y_2)^2}\)
4. 点是否在线段区域
    - 向量 \(\vec{ap}\) 和 向量 \(\vec{ab}\) 的内积小于0， 则P不在线段ab区域内
    - 向量 \(\vec{bp}\) 和 向量 \(\vec{ba}\) 的内积大于等于0，则P在线段ab区域内
5. 点在线段上
    - \(|\vec{ba} \times \vec{bp}| =  |\vec{ba}||\vec{bp}|\sin0 = 0\)， 如果 > eps，则不在线段上
    - < eps 说明在一条直线上，如果p的x，y坐标都在a、b之间，则它在线段上
6. 点到直线的距离
    - \(S = \frac{1}{2}|\vec{pa}||\vec{pb}|\) = 向量叉积的绝对值 / 2
    - \(S = \frac{1}{2}|\vec{ab}|h\)
    - 所以 \(h = \frac{|\vec{pa}||\vec{pb}|}{|\vec{ab}|}\)
7. 线段是否相交
    - 如果某条线段的最小x坐标比另一条最大的x坐标还大，则必然不会相交，同理对y坐标
    - \(\vec{tasa}\times\vec{tatb}\)  与 \(\vec{tatb}\ \times \vec{tasb}\) 同向，内积得pro。 如果pro小于0，则线段不相交；如果pro=0，则sa或sb在tatb上
    - 同理对sa进行与ta相似的操作
8. 点是否在多边形内
    - 对每一条边进行如下操作
    - 1.是否在这条边上
    - 2.从a出发向左的射线与该边无交点不处理
    - 3.射线与边的端点相交，取交点的端点并检查它是否在射线上，如果在count++; 
    - 4.正常相交 count++;
    - 如果最终只有一个相交，则a在内，如果两个相交或者没有相交，则a不在多边形内。

9. 线段是否在多边形内
    - 首先检查该线段是否与某条边相交，相交则返回2
    - 检查a、b是否在多边形内，如果有一个在内部，则返回1
    - 如果两个都在边上，则取中点判断，如果中点在内部，则返回1，如果在边上，则返回-1.
10. 点到多边形的距离
    - 如果a在边上，则返回0.0
    - 扫描a到每个顶点的距离，如果小于min，则为min赋值
    - 如果a在边的区域内，则求a到此边的距离，若小于min，则修改min的值

```cpp
#include <cstdio>

struct Pos{double x,y;} pt[1000005];

            scanf("%lf %lf", &pt[i].x, &pt[i].y);
        }
        pt[n].x = pt[0].x;
        pt[n].y = pt[0].y;

            area += pt[i].x * pt[i+1].y - pt[i].y * pt[i+1].x;
            cx += (pt[i].x + pt[i+1].x) * (pt[i].x * pt[i+1].y - pt[i].y * pt[i+1].x);
            cy += (pt[i].y + pt[i+1].y) * (pt[i].x * pt[i+1].y - pt[i].y * pt[i+1].x);
        }
        printf("%.2lf %.2lf\n", cx / (3 * area), cy / (3 * area));
    }
    return 0;
```

## 4.5 求外接圆圆心
```cpp
#include <cstdio>

using namespace std;
typedef double D;
D GetRadiusBy3Points(D x1, D y1, D x2, D y2, D x3, D y3, D &x, D &y) {
  // 外接圆圆心
  // 由（x-x1)^2 + (y-y1)^2 = (x-x2)^2 + (y-y2)^2得
  // 2*(x2-x1) *x + 2*(y2-y1)*y = x2^2 - x1^2 + y2^2 - y1^2
  // 同理得
  // 2*(x3-x2)*x + 2*(y3-y2)*y = x3^2 - x2^2 + y3^2 - y2^2
  // 由行列式解方程得 x,y
  D a11, a12, a21, a22, b1, b2, d, d1, d2;
  a11 = 2 * (x3 - x2);
  a12 = 2 * (y3 - y2);
  a21 = 2 * (x2 - x1);
  a22 = 2 * (y2 - y1);
  b1 = x3 * x3 - x2 * x2 + y3 * y3 - y2 * y2;
  b2 = x2 * x2 - x1 * x1 + y2 * y2 - y1 * y1;
  d = a11 * a22 - a12 * a21;
  d1 = b1 * a22 - b2 * a12;
  d2 = a11 * b2 - a21 * b1;
  x = d1 / d;
  y = d2 / d;
  return (x1 - x) * (x1 - x) + (y1 - y) * (y1 - y);
}
```

## 4.8 求最远点对
```cpp
#include <cstdio>

const int N = 500;
struct ver {
    int x, y;
};
int cross(ver & aa, ver & bb, ver &cc) {
    return (bb.x - aa.x) * (cc.y - aa.y) - (cc.x - aa.x) * (bb.y - aa.y);
}

bool cmp(const ver & aa, const ver & bb) {
    return aa.y < bb.y || (aa.y == bb.y && aa.x < bb.y);

void rotatingCalipers();

    return sqrt((bb.x - aa.x) *(bb.x - aa.x) + (bb.y - aa.y) *(bb.y - aa.y));

    return ((bb.x - aa.x) * (cc.x - bb.x) + (bb.y - aa.y) * (cc.y - bb.y)) / (dis(aa, bb) * dis(bb, cc));
```

```cpp
void tript(const node & a, const node & b, const node & c);

    // 当点集的点数不超过2的情况下
    if (n < 3) {
        // 如果只有一个点，则最小包围圆是一个点
        if(n == 1) {
            r = 0.0;
            o.x = nd[0].x;
            o.y = nd[0].y;
            return;
        } else {
            // 如果只有两个点，则最小包围圆是以这两个点为直径的圆
            bipt(nd[0], nd[1]);
            return;
        }
    }
    // 如果有多于2个点
    // 首先以前两个点为直径的端点，构造当前最小包围圆
    bipt(nd[0], nd[1]);
    // 从第三个点开始扫描
    for(int i = 2;i < n;i++) {
        // 如果点 i 在当前最小包围圆内部，则continue
        if(incircle(nd[i])) continue;
        // 否则，以点i 和 点 0 为直径端点，构造当前最小包围圆
        bipt(nd[i], nd[0]);
        // 从第2个点扫描到第i - 1 个点
        for(int j = 1;j < i;j++) {
            // 如果点j在当前最小包围圆里，则continue
            if(incircle(nd[j])) continue;
            // 否则，以点j, 点i 和点0 为顶点，构造当前最小包围圆
            tript(nd[i], nd[j], nd[0]);
            // 从第二个点到第j-1个点开始扫描
            for(int k = 1; k < j;k++) {
                // 如果点k到当前最小包围圆里， 则continue
                if(incircle(nd[k])) continue;
                // 否则，以点k, 点i 和点j 为顶点，构造当前最小包围圆
                tript(nd[i], nd[j], nd[k]);
            }
        }
    }
    printf("%.2lf%.2lf%.2lf\n", o.x, o.y, r);
    return;
}

bool incircle(const node & a) {
    double d;
    d = dis(a, o);
    // 如果当前点与当前圆心的距离大于当前半径，则当前点不在内部
    if(d - r > EPS) return false;
    // 否则，在内部
    return true;
}

void bipt(const node & a, const node & b) {
    // 半径为两点间距离的一半
    r = dis(a, b) / 2.0;
    // 坐标为两点中点
    o.x = (a.x + b.x) / 2.0;
    o.y = (a.y + b.y) / 2.0;
    return;
}

void tript(const node & a, const node & b, const node & c) {
    double d1, d2, d3;
    d1 = dis(a, b);
    d2 = dis(a, c);
    d3 = dis(b, c);
    // 如果三点共线，则半径为最长线段中点，圆心为最长线段重点
    if(d1 + d2 < d3 + EPS) {
        r = d3 / 2;
        o.x = (b.x + c.x) / 2;
        o.y = (b.y + c.y) / 2;
    }else if(d1 + d3 < d2 + EPS) {
        r = d2 / 2;
        o.x = (a.x + c.x) / 2;
        o.y = (a.y + c.y) / 2;
    }else if(d2 + d3 < d1 + EPS) {
        r = d1 / 2;
        o.x = (a.x + b.x) / 2;
        o.y = (a.y + b.y) / 2;
    }else { // 否则， 用radisu 函数找半径 r 和圆心 o
        r = radius(a, b, c);
    }
    return;
}

double radius(const node & a, const node & b, const node & c) {
    double a1, a2, b1, b2, c1, c2, d1, d2, d;
    a1 = 2 * (c.x - b.x), a2 = 2 * (c.y - b.y);
    b1 = 2 * (b.x - a.x), b2 = 2 * (b.y - a.y);
    c1 = c.x * c.x - b.x * b.x + c.y * c.y - b.y * b.y;
    c2 = b.x * b.x - a.x * a.x + b.y * b.y - a.y * a.y;
    d = a1 * b2 - a2 * b1;
    d1 = c1 * b2 - c2 * a2;
    d2 = a1 * c2 - c1 * b1;
    // 根据几何公式推出半径r 和圆心 o
    o.x = d1/ d;
    o.y = d2/d;
    return sqrt((a.x - o.x) * (a.x - o.x) + (a.y - o.y) * (a.y - o.y));
}
```

```cpp
    while(scanf("%d%lf", &n, &r) != EOF) {
        init();
        sol();
    }
    return 0;
}

void init() {
    db alpha, dx, dy;
    int sigdx, sigdy;
    for(int i = 1;i <= n;i++) {
        scanf("%lf%lf", &pt[i].x, &pt[i].y);
    }
    for(int i = 1;i <= n;i++) {
        int j = i == n?1:i+1;
        dx = pt[j].x - pt[i].x;
        dy = pt[j].y - pt[j].y;
        sigdx = sig(dx);
        sigdy = sig(dy);
        if(sigdx == 0) {
            if(sigdy < 0) alpha = PI/2;
            else alpha = -PI/2;
        } else if(sigdy == 0) {
            if(sigdx < 0) alpha = 0.0;
            else alpha = PI;
        } else {
            dx = fabs(dx);
            dy = fabs(dy);
            if(sigdx > 0) {
                if(sigdy > 0) alpha = PI + atan(dy/dx);
                else alpha = PI - atan(dy/dx);
            } else {
                if(sigdy > 0) alpha = -atan(dy/dx);
                else alpha = atan(dy/dx);
            }
        }
        seg[i].a.x = pt[j].x - sin(alpha) * r;
        seg[i].a.y = pt[j].y + cos(alpha) * r;
        seg[i].b.x = pt[j].x - sin(alpha) * r;
        seg[i].b.y = pt[j].y + cos(alpha) * r;
    }
    sort(seg + 1, seg+ n+1, cmp);
    return;
}

void sol() {
    int topn, flag, topp;
    db d1, d2;
    // maxd 和 tempd 只是这道题用
    db maxd, tempd;
    Point temp;
    topn = 2;
    for(int i = 2;i <= n;i++) {
        d1 = atan2(seg[i-1].b.y - seg[i-1].a.y, seg[i-1].b.x - seg[i-1].a.x);
        d2 = atan2(seg[i].b.y - seg[i].a.y, seg[i].b.x - seg[i].a.x);
        if(sig(d1 - d2) == 0) continue;
        seg[topn++] = seg[i];
    }
    n = topn - 1;
    que[0] = 1;
    que[1] = 2;
    qs = 0, qe = 2;
    for(int i = 3;i <= n;i++) {
        while(1) {
            lineIntersect(seg[que[qe-1]].a, seg[que[qe-1]].b, seg[que[qe-2]].a, seg[que[qe-2]].b, temp.x, temp.y);
            if(!onLeft(seg[i].a, seg[i].b, temp)) {
                qe--;
                if(qe-qs < 2) break;
            } else {
                break;
            }
        }
        while(1) {
            if(qe-qs < 2) break;
            lineIntersect(seg[que[qs]].a, seg[que[qs]].b, seg[que[qs+1]].a, seg[que[qs+1]].b, temp.x, temp.y);
            if(!onLeft(seg[i].a, seg[i].b, temp)) {
                qs++;
            } else {
                break;
            }
        }
        que[qe++] = i;
    }
    while(1) {
        flag = 0;
        while(1) {
            lineIntersect(seg[que[qe-1]].a, seg[que[qe-1]].b, seg[que[qe-2]].a, seg[que[qe-2]].b, temp.x, temp.y);
            if(!onLeft(seg[que[qs]].a, seg[que[qs]].b, temp)) {
                qe--;flag = 1;
            } else {
                break;
            }
        }
        while(1) {
            lineIntersect(seg[que[qs]].a, seg[que[qs]].b, seg[que[qs+1]].a, seg[que[qs+1]].b, temp.x, temp.y);
            if(!onLeft(seg[que[qe-1]].a, seg[que[qe-1]].b, temp)) {
                qs++;flag = 1;
            } else {
                break;
            }
        }
        if(!flag) break;
    }
    que[qe] = que[qs];
    topp = 0;
    for(int i = qs;i < qe;i++) {
        lineIntersect(seg[que[i]].a, seg[que[i]].b, seg[que[i+1]].a, seg[que[i+1]].b, temp.x, temp.y);
        poly[topp++] = temp;
    }
    poly[topp] = poly[0];
    maxd = 0.0;
    o1 = o2 = 0;
    for(int i = 0;i < topp; i++) {
        for (int j = i+1; j < topp;j++) {
            tempd = dis(i,j);
            if(sig(maxd-tempd) <= 0) {
                maxd = tempd;
                o1 = i;
                o2 = j;
            }
        }
    }
    if(fabs(poly[o1].x) < NZ) poly[o1].x = 0.0;
    if(fabs(poly[o1].y) < NZ) poly[o1].y = 0.0;
    if(fabs(poly[o2].x) < NZ) poly[o2].x = 0.0;
    if(fabs(poly[o2].y) < NZ) poly[o2].y = 0.0;
    printf("%.4lf %.4lf %.4lf %.4lf\n", poly[o1].x, poly[o1].y, poly[o2].x, poly[o2].y);
    return;
}
```
