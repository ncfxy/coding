```cpp
        temp = INF;

                temp = dis[j];
```

```cpp
        d[i].next = NULL;
        dist[i] = INF;

        pool[top].w = w;

        pool[top].w = w;

                dist[st->to] = dist[flag] + st->w;
            }
            st = st->next;
        }
    }
```

## 3.3 Dijkstra(O(nlogn))
```cpp
/* 求无负权有向图的最短路径的dijkstra算法，用二叉堆实现
时间复杂度O((n+m)logn)
by ftfish, 07.05.30
*/
#include <cstdio>
#include <list>
using namespace std;

const int mn = 1001, inf = 1<<30;
int n, m, s, t, dist[mn], p[mn], h[mn], pos[mn];
struct lnode{int v,w;};
list<lnode> g[mn];
inline void swaph(int a, int b) {swap(h[a], h[b]}; pos[h[a]] = a; pos[h[b]] = b;}
void init();
void shiftup(int p);
int delmin();
void dijkstra();
void showpath(int id);
void output();
int main(){
    while(scanf("%d%d", &n, &m) != EOF){
        init();
        dijkstra();
        output();
    }
    return 0;
}
void init(){
    s = ;
    t = ;
    for(int i = 1; i <= n; i++){
        g[i].clear();
        d[i] = inf;
        h[i] = pos[i] = i;
        p[i] = 0;
    }
    d[s] = 0;
    hs = n;
    swaph(1,s);
    for(int i = 1, a,b,c; i <= m;i+++){
        scanf("%d%d%d", &a, &b, &c);
        g[a].push_back((lnode){b,c});
        // g[b].push_back((lnode){a,c});    // 加上此句为无向图
    }
}
void shiftup(int p){
    for(int i = p;i > 1&& d[h[i/2]] > d[h[i]];i /= 2) swaph(i, i/2);
}
int delmin(){
    int ret = h[1];
    swaph(1, hs--);
    for(int i = 1, tmp; i <= hs/2;i = tmp){
        tmp = i*2;
        if(tmp < hs && d[h[tmp+1]] < d[h[tmp]]) tmp++;
        if(d[h[i] <= d[h[tmp]]]) break;
        swaph(i, tmp);
    }
    return ret;
}
void dijkstra(){
    while(hs) {
        int mp = delmin();
        if(mp==t)return;    // 去掉此句得到s到所有点的距离
        for(list<lnode>::iterator i = g[mp].begin(); i != g[mp].end(); i++){
            if(pos[i->v] <= hs && d[i->v] > d[mp] + i->w){
                d[i->v] = d[mp] + i->w;
                p[i->v] = mp;
                shiftup(pos[i->v]);
            }
        }
    }
}
void showpath(int n){
    if(p[n]) {
        showpath(p[n]);
        putchar(' ');
    }
    printf("%d", n);
}
void output(){
    if(d[t] == inf){
        printf("No way!");
        return;
    }
    printf("Distance between (%d,%d) is %d\n", s, t, d[t]);
    showpath(t);
    putchar('\n');
}
```

## 3.4 Bellman_Ford O(VE)
```cpp
// 在建边的时候注意，如果是单向边直接建边，如果是双向边，而且是类似与wormholes
// 的类型，讨论边是否已经存在 tju 2831 0.04s n = 500 m= 2500 w =200
#include <cstdio>
#include <string>
#include <string.h>
const int M = 1005; // M 记录点数上限
const int N = 100005;   // N 记录边数上限，如果是无向边，建2*N
const int INF = 1 << 30;
struct node {
    int from, to, w;
}edge[2 * N];
int n;  // 记录点数
int m;  // 记录边数，如果是无向边，m *=2;
int mm; // 记录负边数（如果题目输入是这样的话）
int s;  // 记录起点
int dist[M]; // 记录点到s点的距离
void init();
bool Bellman_Ford();
int main() {
    int t;
    scanf("%d", &t);
    while(t--) {
        init();
        if(Bellman_Ford()) {
            printf("No negative ring(s).\n");
        } else {
            printf("Has Negative ring(s).\n");
        }
    }
    return 0;
}
void init() {
    int u,v,w;
    scanf("%d%d%d", &n, &m, &mm);
    s = 1;
    for(int i = 1;i <= n;i++) dist[i] = INF;
    for(int i = 0;i < m;i++) {
        scanf("%d%d%d", &u, &v, &w);
        edge[i].from = u;
        edge[i].to = v;
        edge[i].w = w;
        edge[i + m].from = v;
        edge[i + m].to = u;
        edge[i + m].w = w;
    }
    dist[s] = 0;
    for(int i = m;i <m + mm;i++){
        scanf("%d%d%d", &u, &v, &w);
        edge[i].from = u;
        edge[i].to = v;
        edge[i].w = -w;
    }
    m+=mm;
    return;
}
bool Bellman_Ford() {
    int u,v,w,flag;
    for(int i = 1;i <= n;i++) {
        flag = 0;
        for(int j = 0;j < m;j++) {
            u = edge[j].from;
            v = edge[j].to;
            w = edge[j].w;
            if(dist[v] > dist[u] + w) {
                dist[v] = dist[u] + w;
                flag = 1;
            }
        }
        if(!flag) break;
        if(i == n && flag) return false;
    }
    return true;
}
```

```cpp
#include <string>
#include <string.h>
const int M = 205;
const int INF = 1 << 28;
int n; // 点数
int m;  // 边数
int d[M][M];    // 记录边权
void init();
void Floyd();

    while(scanf("%d%d", &n, &m) != EOF){
        init();
        Floyd();
    }
    return 0;
}
void init(){
    int u,v,w;
    for(int i = 1;i <= n;i++){
        for(int j = 1;j <= n;j++){
            d[i][j] = i == j ? 0 : INF;
        }
    }
    for(int i = 0;i < m;i++){
        scanf("%d%d%d", &u, &v, &w);

        d[v][u] = d[u][v];

void Floyd(){
    for(int k = 1;k <= n;k++){
        for(int i = 1;i <= n;i++){
            for(int j = 1;j <= n;j++){
                if(d[i][j] > d[i][k] + d[k][j]){
                    d[i][j] = d[i][k] + d[k][j];
                }
            }
        }
    }
    return;
}
```

## Prim(O(nlogn))
```cpp
/*
求无负权有向图的最短路径的Dijkstra算法， 用二叉堆实现
时间复杂度O(mlogn)
*/
#include <cstdio>
#include <list>
using namespace std;
const int mn = 1001, inf = 1 << 30;
int n, m, s, res, cnt, hs, d[mn], p[mn], h[mn], pos[mn];
struct lnode{ int v, w;};
list<lnode> g[mn];
inline void swaph(int a, int b) {
    swap(h[a], h[b]);
    pos[h[a]] = a;
    pos[h[b]] = b;
}

void init() {
    scanf("%d%d", &n, &m);
    res = 0, cnt = 0;
    for (int i = 1;i <= n; i++) {
        g[i].clear();
        d[i] = inf;
        h[i] = pos[i] = i;
        p[i] = 0;
    }
    d[s] = 0; 
    hs = n;
    s = 1;
    swaph(1, s);
    for (int i = 1, a,b,c; i <= m; i++) {
        scanf("%d%d%d", &a, &b, &c);
        g[a].push_back({b, c});
        // g[b].push_back((lnode){a, c}); 加上此句为无向图
    }
    return;
}

void shiftup(int p) {
    for (int i = p; i > 1 && d[h[i]] < d[h[i/2]]; i /= 2) {
        swaph(i, i/2);
    }
    return;
}

int delmin() {
    int ret = h[1];
    swaph(1, hs--);
    for (int i =1, tmp; i <= hs/2;i = tmp) {
        tmp = i * 2;
        if (tmp < hs && d[h[tmp+1]] < d[h[tmp]]) tmp++;
        if (d[h[i]] <= d[h[tmp]]) break;
        swaph(i, tmp);
    }
    return ret;
}

void prim() {
    while(hs) {
        int mp = delmin();
        if (d[mp] < inf) {
            res += d[mp];
            cnt++;
        }
        d[mp] = 0;
        for (list<lnode>::iterator it = g[mp].begin(); it != g[mp].end(); it++) {
            if (pos[it->v] <= hs && d[it->v] > d[mp] + it->w) {
                d[it->v] = d[mp] + it->w;
                p[it->v] = mp;
                shiftup(pos[it->v]);
            }
        }
    }
    return;
}

void output() {
    if (cnt < n) {
        printf("No way!");
        return;
    }
    printf("Distance is %d\n", res);
}

int main() {
    init();
    prim();
    output();
    return 0;
}
```

## Kruskal O(E * logE)
```cpp
// 注意边数上限不能太小

#include <cstdio>
#include <string>
#include <algorithm>

using namespace std;

const int M = 1005;
const int N = 10005;

struct node {
    int from;
    int to;
    int w;
}edge[N];  // from 起点，to 终点，w 权值

int n, m, res;
int fa[M];
int rk[M];

void INIT();
void sol();
int Find(int k);
void Union(int x, int y);
bool Kruskal();

bool cmp(const node &a, const node &b) {
    return a.w < b.w;
}

int main() {
    while (scanf("%d%d", &n, &m) != EOF) {
        INIT();
        sol();
    }
    return 0;
}

void INIT() {
    int u, v, w;
    for (int i = 0;i < m; i++) {
        scanf("%d%d%d", &u, &v, &w);
        edge[i].from = u;
        edge[i].to = v;
        edge[i].w = w;
    }
    for (int i = 1;i <= n; i++) {
        fa[i] = i;
        rk[i] = 0;
    }
    res = 0;
    m *= 2;
    sort(edge, edge + m, cmp);
    return;
}

void sol() {
    if(Kruskal()) {
        printf("%d\n", res);
    }else {
        printf("Fail to construct a connected map.\n");
    }
    return;
}

int Find(int x) {
    if (x == fa[x]) return x;
    else return fa[x] = Find(fa[x]);
}

void Union(int x, int y) {
    x = Find(x);
    y = Find(y);
    if (x != y) {
        if (rk[x] < rk[y]) fa[x] = y;
        else if (rk[x] > rk[y]) fa[y] = x;
        else {
            fa[x] = y;
            rk[y]++;
        }
    }
    return;
}

bool Kruskal() {
    for (int i = 0;i < m; i++) {
        if (Find(edge[i].from) != Find(edge[i].to)) {
            Union(edge[i].from, edge[i].to);
            res += edge[i].w;
        }
    }
    for (int i = 2;i <= n; i++) {
        if (Find(i) != Find(1)) return false;
    }
    return true;
}
```

## 3.10 缩点
```cpp
// 缩点，两遍DFS
#include <cstdio>
#include <string>
#include <string.h>
#include <vector>

using namespace std;

const int M = 10005;
const int INF = 1 << 30;
int n, m; // n 点数，m 边数
int top; // 记录队列
int mark; // mark记录缩点之后的点
vector<int> Map[M]; // 记录原图
vector<int> un_Map[M]; // 记录反向图
vector<int> daMap[M]; // 记录sec图
int cover[M]; // 记录每个点是否被覆盖
int queue[M]; // 记录反向图遍历的顺序
int belong[M]; // 记录缩点之后的新点

void INIT();
void sol();
void sec();// 用于两遍DFS, 缩点
void compress(); // 用于压缩点，求sec图
void DFS(int k); 
void DFS_2(int k);

int main() {
    while (scanf("%d%d", &n, &m) != EOF) {
        if (n == 0 && m == 0) break;
        INIT();
        sol();
    }
    return 0;
}

void INIT() {
    int u, v;
    top = 0; mark = 0;
    for (int i = 1;i <= n; i++) {
        Map[i].clear();
        un_Map[i].clear();
        daMap[i].clear();
    }
    for (int i = 0;i < m; i++) {
        scanf("%d%d", &u, &v);
        Map[u].push_back(v);
        un_Map[v].push_back(u);
    }
    return;
}

void sol() {
    sec();
    compress();
    return;
}

void sec() {
    memset(cover, 0, sizeof(cover));
    for (int i = 1;i <= n; i++) {
        if (!cover[i]) DFS(i);
    }
    memset(cover, 0, sizeof(cover));
    for (int i = top;i > 0; i--) {
        if (!cover[queue[i]]) {
            mark++;
            DFS_2(queue[i]);
        }
    }
    return;
}

void compress() {
    for (int i = 1;i <= n; i++) {
        for (int j = 0;j < Map[i].size(); j++) {
            if (belong[i] != belong[Map[i][j]]) {
                daMap[belong[i]].push_back(belong[Map[i][j]]);
            }
        }
    }
    return;
}

void DFS(int k) {
    cover[k] = 1;
    for (int i = 0;i < Map[k].size(); i++) {
        if (!cover[Map[k][i]]) DFS(Map[k][i]);
    }
    cover[k] = 2;
    queue[++top] = k;
    return;
```

## 3.14 Difference Constraint System(差分约束系统)
```cpp
/* 差分约束系统
将不等式转化成图来求解
1. 如果要求某个值的最大值，在建图的时候，就要把所有的不等式转化成这样的方程:
 A - B <= C 之后添加边: B -> A, 权值为 C。 dist初始化为正无穷，求最短路
2. 如果要求某个值的最小值，在建图的时候，就要把所有的不等式转化成这样的方程:
    A - B >= C 之后添加边: A -> B, 权值为 C。 dist初始化为负无穷，求最长路
有负环说明无解，否则解为dist数组的值
*/
#include <cstdio>
#include <string>

const int M = 50005;
const int INF = 100000005;

struct node {
    int u, v, c;
} edge[M *4];

int dist[M];
bool Bellman(int s, int n, int m);

int main() {
    int n, m, s, res;
    while(scanf("%d", &m) != EOF) {
        n = 0;
        for (int i = 0;i < m; i++) {
            scanf("%d %d %d", &edge[i].u, &edge[i].v, &edge[i].c);
            edge[i].v = edge[i].u - 1;
            edge[i].c = -edge[i].c;
            n = n > edge[i].u ? n : edge[i].u;
        }
        s = n + 1;
        for (int i = m; i < m + n; i++) {
            edge[i].v = i - m;
            edge[i].u = i - m + 1;
            edge[i].c = 0;
            edge[i+n].v = i - m + 1;
            edge[i+n].u = i - m;
            edge[i+n].c = 1;
        }
        for (int i = m + 2 * n; i <= m + 2 * n + n; i++) {
            edge[i].u = s;
            edge[i].v = i - (m + 2 * n);
            edge[i].c = 0;
        }
        if (Bellman(s, n + 2, m + 3 * n  + 1)) {
            res = dist[n];
            if (dist[0] < 0) res -= dist[0];
            printf("%d\n", res);
        }
    }
    return 0;
}

bool Bellman(int s, int n, int m) {
    int u, v, c, flag;
    for(int i = 0;i < n;i++) dist[i] = INF;
    dist[s] = 0;
    for(int i = 1;i <= n;i++) {
        flag = 0;

            u = edge[j].u;
            v = edge[j].v;

                dist[v] = dist[u] + c;
                flag = 1;
            }
        }
        if (flag == 0) break;
        if (i == n && flag == 1) return false;
    }
    return true;
```

```cpp
int q[M]; // 记录待处理右节点队列

        d[u][v] = true;

int Bipartite() {
    int x, qs, qe, r;
    for (int i = 1;i <= n1; i++) v1[i] = -1;
    for (int j = 1; j<= n2; j++) v2[j] = -1;  // 初始化所有点为未被匹配的状态
    r = 0;
    for (int i = 1;i <= n1; i++) {
        if (v1[i] != -1) continue;   // 对于左边每一个未被匹配的点进行一次bfs找交错链
        for (int j = 1; j <= n2; j++) prev[j] = -1; // 每次bfs初始化右边的点
        qs = qe = 0; // 初始化bfs队列
        for (int j = 1; j <= n2; j++) {
            if (d[i][j]) {
                prev[j] = -1;
                q[qe++] = j;
            }
        } // 从初始化的那个点开始，先把它能找的右边的点放入队列
        while(qs < qe) {
            x = q[qs];
            if (v2[x] == -1) break;
            qs++;
            for(int j = 1;j <= n2; j++) {
                if (prev[j] == -2 && d[v2[x]][j]) {  // 扩展结点
                    prev[j] = x;  // 如果该右边点事一个已经匹配的点，则v2[x]是与它相匹配的点从该左边点出发，寻找其他可以找到的右边点
                    q[qe++] = j;
                }
            }
        }
        if (qs == qe) continue;  // 没有找到交错链
        while(prev[x] > -1) {
            v1[v2[prev[x]]] = x;
            v2[x] = v2[prev[x]];
            x = prev[x];
        }
        v1[i] = x; v2[x] = i;r++;
    }
    return r;
```

## 3.18 KM 算法（带权二分图）
```cpp

// 如果求最小权匹配，只需要把边权改为负
// X集合和Y集合要相等，如果不相等，加点添0边

#include <string.h>

            w[i][j] = -INF;

        for (int k = 1; k <= n;k++){

    printf("Case %d: %d\n", cas++, -res);

int KM() {

    ans = 0;
    memset(y, 0, sizeof(y));
    for (int i = 1;i <= n;i++) {
        x[i] = -INF;
        for(int j = 1;j <= n;j++)
            if(w[i][j] > x[i])
                x[i] = w[i][j];
    }
    memset(prev, 0, sizeof(prev));
    for(int i = 1;i <= n;i++) {
        memset(slack, 0x3f, sizeof(slack));
        memset(coverx, 0, sizeof(coverx));
        memset(covery, 0, sizeof(covery));
        while(!path(i)) {
            d = INF;
            for (int j = 1;j <= n;j++) 
                if(!covery[j]) d = min(d,slack[j]);
            for(int k = 1; k <= n;k++) {
                if(coverx[k]) {
                    x[k] -= d;
                    coverx[k] = 0;
                }
                if (covery[k]) {
                    y[k] += d;
                    covery[k] = 0;
                }else {
                    slack[k] -= d;
                }
            }
        }
    }
    for (int i = 1;i <= n;i++) {
        ans += w[prev[i]][i];
    }
    return ans;
}

    int temp;
    coverx[i] = 1;
    for(int j = 1;j <= n;j++) {
        if (covery[j]) continue;
        temp = x[i] + y[j] - w[i][j];
        if(temp == 0) {
            covery[j] = 1;
            if(prev[j] == 0 || path(prev[j])) {
                prev[j] = i;
                return 1;
            }
        }
        else if(slack[j] > temp) {
            slack[j] = temp;
        }
    }

    return 0;
```

```cpp
void INIT() {
    int u, v, w;
    memset(g, 0, sizeof(g));
    memset(f, 0, sizeof(f));
    s = n+1; t = n+2; F = 0;    // 增加源点和汇点
    for(int i = 1;i <= m;i++) {
        scanf("%d%d%d", &u, &v, &w);
        g[u][v] = w;
    }
    n += 2;
    return;

void MaxFlow() {
    if (BFS_Find_Path()) argument();
    return;

bool BFS_Find_Path() {
    int qs, qe, u, v;
    memset(num, 0, sizeof(num));
    for(int i = 1;i <= n;i++)d[i]= n; // d[i] 表示i点到t点的距离，相当于常用的distance
    d[t] = 0;
    num[0] = 1;
    num[n] = n-1;
    qs = qe = 0;
    q[qe++] = t;
    while(qs < qe) {
        v = q[qs++];
        for(u = 1; u <= n; u++) {
            if (g[u][v] > 0 && d[u] == n) { // 他的BFS是倒着从t找回s，所以是g[u][v]
                d[u] = d[v] + 1;
                q[qe++] = u;
                num[n]--;
                num[d[u]]++;  // num[i] 表示到起始点距离为i的点的个数
            }
        }
    }
    if(d[s] == n) return false;  // d[s] == n 表示没有找到增广路
    return true;
}

void argument() {
    int u, v, ex;
    int td, tnd;
    memset(fa, -1, sizeof(fa));
    for (int i = 1;i <= n;i++) cnode[i] = 1;  // cnode[] 表示后继节点，fa表示前驱节点。
    u = s;
    while(d[s] < n) {
        for(v = cnode[u];v <= n;v++) if (g[u][v] > 0 && d[u]==d[v]+1)break;
        if(v <= n) {
            cnode[u] = v;
            fa[v] = u;
            u = v;
            if (v == t) {
                ex = 1 << 30;
                for(v = t; v!=s;v=fa[v])if(ex > g[fa[v]][v])ex = g[fa[v]][v]; // 这条路能支持的最大流，即可增广的流
                for(v = t;v!=s;v = fa[v]) {
                    f[fa[v]][v] += ex; // 正向流量增加，逆向减少，保持网络流反对称性
                    f[v][fa[v]] -= ex;
                    g[fa[v]][v] -= ex;
                    g[v][fa[v]] += ex;
                }
                F += ex;
                u = s; // 把u重新赋值为s，从BFS的结果中记录找增广路
            }
        } else { // 已有结果中再无增广路，当u没有可走的边时，则降d[u]的距离标号修改为可到的最小的d[j]+1，其中j为u的邻接点
            cnode[u] = 1;
            td = d[u];
            tnd = n - 1;
            for(v = 1; v <= n; v++) if(g[u][v] > 0 && tnd > d[v]) tnd = d[v];
            num[td]--;
            num[d[u] = tnd+1]++;        // 如果换有与u相连的点，把u更新
            if(num[td] == 0)break;      // td数量为0时，已不可能再有路径
            if(u != s) u = fa[u];
        }
    }
    return;
}
```

```cpp
        pass[id] = 1;

            cover[q[i]] = 1;

                d[id][j] = d[q[i]][j];

            for(j = from; j < top;j++) {
                k = q[j];
                if(d[i][id] > d[i][k] - d[in[k]][k])
                    d[i][id] = d[i][k] - d[in[k]][k];
            }
        }
    }
    return;
}
```

```cpp
void lim() {
  int num = 0, sub;
  res = 0;
  for (int i = 0; i < n; i++)
    cover[i] = 0;
  memset(used, 0, sizeof(used));
  memset(head, -1, sizeof(head));
  cnt = 0;
  for (int i = 0; i < n; i++) {
    if (!cover[i] && i != pt) {
      ptt = i;
      res += prim();
      num++;
    }
  }
  if (num > k) {
    res = -1;
    return;
  }
  while (num < k) {
    bfs();
    sub = add_degree(n, ptt);
    if (sub > 0)
      break;
    res += sub;
    num++;
  }
  return;
}

void bfs() {
  int top, eid, u, v;
  top = 0;
  q[top++] = pt;
  dp[pt] = -1;
  eid = head[pt];
  memset(vis, 0, sizeof(vis));
  vis[pt] = 1;
  while (eid != -1) {
    v = e[eid].v;
    if (!vis[v]) {
      vis[v] = 1;
      dp[v] = -1;
      q[top++] = v;
    }
    eid = e[eid].next;
  }
  for (int i = 0; i < top; i++) {
    u = q[i];
    eid = head[u];
    while (eid != -1) {
      v = e[eid].v;
      if (used[u][v] && vis[v] == 0) {
        vis[v] = 1;
        q[top++] = v;
        dp[v] = dp[u];
        da[v] = da[u] + 1;
        db[v] = db[u];
        if (d[u][v] > dp[v]) {
          dp[v] = d[u][v];
          da[v] = u;
          db[v] = v;
        }
      }
      eid = e[eid].next;
    }
  }
  return;
}

int add_degree(int tn, int t) {
  int sum = 0, sub = 0, id = 0, u, v;
  for (int i = 0; i < tn; i++) {
    // 注意是 t 不是 pt
    if (i != t && used[i][t] == 0) {
      if (d[i][t] != INF && sub > d[i][t] - dp[i]) {
        sub = d[i][t] - dp[i];
        id = i;
      }
    }
  }
  u = da[id], v = db[id];
  e[cnt].v = id, e[cnt].next = head[t], head[t] = cnt++;
  e[cnt].v = t, e[cnt].next = head[id], head[id] = cnt++;
  d[u][v] = d[v][u] = 0;
  d[t][id] = d[id][t] = 1;
  return sub;
}

int prim() {
  // 注意一个 tmin, 一个mmin
  int tmin, id, mmin, pp, sum = 0;
  for (int i = 0; i < n; i++) {
    low[i] = d[ptt][i];
    pre[i] = ptt;
  }
  cover[ptt] = 1, tmin = INF, id = -1;
  tmin = INF, id = -1;
  if (d[ptt][pt] != INF) {
    tmin = d[ptt][pt], id = ptt;
  }
  for (int i = 1; i < n; i++) {
    mmin = INF, pp = -1;
    for (int j = 0; i < n; j++)
      if (!cover[j] && j != pt && mmin > low[j]) {
        mmin = low[j], pp = j;
      }
    if (pp == -1)
      break;
    sum += mmin, cover[pp] = 1;
    used[pp][pre[pp]] = used[pre[pp]][pp] = 1;
    e[cnt].v = pre[pp], e[cnt].next = head[pp], head[pp] = cnt++;
    e[cnt].v = pp, e[cnt].next = head[pre[pp]], head[pre[pp]] = cnt++;
    for (int j = 0; j < n; j++) {
      if (cover[j] && j != pt && d[pp][j] != INF && d[pp][j] < low[j]) {
        low[j] = d[pp][j];
        pre[j] = pp;
      }
    }
  }
  if (id == -1)
    return -1;
  e[cnt].v = id, e[cnt].next = head[pt], head[pt] = cnt++;
  e[cnt].v = pt, e[cnt].next = head[id], head[id] = cnt++;
  used[id][pt] = used[pt][id] = 1;
  sum += tmin;
  return sum;
}
```

```cpp
    double sum = 0, mm;
    int i, pp, j;

        dist[i] = cost[0][i] - r * dis[0][i];
        mark[i] = false;
    }
    mark[0] = true;

                dist[j] = cost[pp][j] - r * dis[pp][j];

    }

}
```

```cpp
// 第 k 短路 by mmd 手写链表
// pku 2449 单向边，如果 s == t k++ 
#include <iostream>
#include <algorithm>

const int maxint = 0x7fffffff;
const int maxn = 1010;

    int v, w;
    _adj *next;
}*adj[maxn], heap[maxm * 10], *adjn[maxn];
int dheap[maxn], dheap_len, dhash[maxn];
int heap_len;
int n, m, k, stt, trm;
int d[maxn];
inline void dheap_pop(int k) {
    dhash[dheap[k] = dheap[dheap_len]] = k;
    dheap[dheap_len--] = 0;
    int i;
    while((i = k << 1) <= dheap_len) {
        if(d[dheap[i+1]] < d[dheap[i]]) i++;
        if(d[dheap[i]] < d[dheap[k]]) {
            swap(dhash[dheap[k]], dhash[dheap[i]]);
            swap(dheap[k], dheap[i]);
            k = i;
        } else break;
    }
}
inline void dheap_up(int k) {
    if(dhash[k] == 0) {
        dheap[dhash[k] = ++dheap_len] = k;
    }
    k = dhash[k];
    int i;
    while(i = k >> 1) {
        if(d[dheap[k]] < d[dheap[i]]) {
            swap(dhash[dheap[k]], dhash[dheap[i]]);
            swap(dheap[k], dheap[i]);
            k = i;
        } else break;
    }
}

void dij(int stt) {
    int v, w;
    _adj *pt;
    for(v = 0;v <= n;v++) {
        d[v] = maxint;
        dhash[v] = 0;
    }
    d[stt] = 0;
    dheap[dheap_len = 1] = stt;
    while(dheap_len) {
        if(d[v = dheap[1]] < maxint) {
            dheap_pop(1);
            for(pt = adjn[v];pt;pt = pt->next) {
                if(d[v] + pt->w < d[w=pt->v]) {
                    d[w] = d[v] + pt->w;
                    dheap_up(w);
                } else {
                    break;
                }
            }
        }
    }
}

bool comp(const _adj &a, const _adj &b) {
    return a.w + d[a.v] > b.w + d[b.v];
}

    dij(trm);

            heap[heap_len].w = w + pt->w;

        }
    }

}
```
