```cpp
            d[u][v] = w;
        }
        d[v][u] = d[u][v];

    memset(cover, 0, sizeof(cover));
    s = ;
    t = ;
    return;
}

    for(int i = 0;i < n;i++){
        temp = INF;
        for(int j = 1;j <= n;j++){
            if(cover[j] == 0 && dis[j] < temp){
                temp = dis[j];
                flag = j;
```

```cpp
        printf("%d\n", dist[t]);

    for(int i = 1; i <= n; i++){
        d[i].next = NULL;
        dist[i] = INF;
    }
    s = ;
    t = ;

    for(int i = 0; i < m; i++){
        scanf("%d%d%d", &u, &v, &w);

        pool[top].w = w;
        pool[top].next = d[u].next;
        d[u].next = &pool[top++];

        pool[top].to = u;
        pool[top].w = w;
        pool[top].next = d[v].next;
        d[v].next = &pool[top++];
    }
    return;

    dist[s] = 0;

        temp = INF;
        for(int j = 1;j <= n;j++){

                temp = dist[j];

        if(flag == t) return;

        while(st != NULL){
            if(cover[st->to] == 0 && dist[st->to] > dist[flag] + st->w){
                dist[st->to] = dist[flag] + st->w;
            }
            st = st->next;
        }
    }
    return;
}
```

## 3.3 Dijkstra(O(nlogn))
```cpp
/* 求无负权有向图的最短路径的dijkstra算法，用二叉堆实现
时间复杂度O((n+m)logn)
by ftfish, 07.05.30
*/
#include <cstdio>
#include <list>
using namespace std;

const int mn = 1001, inf = 1<<30;
int n, m, s, t, dist[mn], p[mn], h[mn], pos[mn];
struct lnode{int v,w;};
list<lnode> g[mn];
inline void swaph(int a, int b) {swap(h[a], h[b]}; pos[h[a]] = a; pos[h[b]] = b;}
void init();
void shiftup(int p);
int delmin();
void dijkstra();
void showpath(int id);
void output();
int main(){
    while(scanf("%d%d", &n, &m) != EOF){
        init();
        dijkstra();
        output();
    }
    return 0;
}
void init(){
    s = ;
    t = ;
    for(int i = 1; i <= n; i++){
        g[i].clear();
        d[i] = inf;
        h[i] = pos[i] = i;
        p[i] = 0;
    }
    d[s] = 0;
    hs = n;
    swaph(1,s);
    for(int i = 1, a,b,c; i <= m;i+++){
        scanf("%d%d%d", &a, &b, &c);
        g[a].push_back((lnode){b,c});
        // g[b].push_back((lnode){a,c});    // 加上此句为无向图
    }
}
void shiftup(int p){
    for(int i = p;i > 1&& d[h[i/2]] > d[h[i]];i /= 2) swaph(i, i/2);
}
int delmin(){
    int ret = h[1];
    swaph(1, hs--);
    for(int i = 1, tmp; i <= hs/2;i = tmp){
        tmp = i*2;
        if(tmp < hs && d[h[tmp+1]] < d[h[tmp]]) tmp++;
        if(d[h[i] <= d[h[tmp]]]) break;
        swaph(i, tmp);
    }
    return ret;
}
void dijkstra(){
    while(hs) {
        int mp = delmin();
        if(mp==t)return;    // 去掉此句得到s到所有点的距离
        for(list<lnode>::iterator i = g[mp].begin(); i != g[mp].end(); i++){
            if(pos[i->v] <= hs && d[i->v] > d[mp] + i->w){
                d[i->v] = d[mp] + i->w;
                p[i->v] = mp;
                shiftup(pos[i->v]);
            }
        }
    }
}
void showpath(int n){
    if(p[n]) {
        showpath(p[n]);
        putchar(' ');
    }
    printf("%d", n);
}
void output(){
    if(d[t] == inf){
        printf("No way!");
        return;
    }
    printf("Distance between (%d,%d) is %d\n", s, t, d[t]);
    showpath(t);
    putchar('\n');
}
```

## 3.4 Bellman_Ford O(VE)
```cpp
// 在建边的时候注意，如果是单向边直接建边，如果是双向边，而且是类似与wormholes
// 的类型，讨论边是否已经存在 tju 2831 0.04s n = 500 m= 2500 w =200
#include <cstdio>
#include <string>
#include <string.h>
const int M = 1005; // M 记录点数上限
const int N = 100005;   // N 记录边数上限，如果是无向边，建2*N
const int INF = 1 << 30;
struct node {
    int from, to, w;
}edge[2 * N];
int n;  // 记录点数
int m;  // 记录边数，如果是无向边，m *=2;
int mm; // 记录负边数（如果题目输入是这样的话）
int s;  // 记录起点
int dist[M]; // 记录点到s点的距离
void init();
bool Bellman_Ford();
int main() {
    int t;
    scanf("%d", &t);
    while(t--) {
        init();
        if(Bellman_Ford()) {
            printf("No negative ring(s).\n");
        } else {
            printf("Has Negative ring(s).\n");
        }
    }
    return 0;
}
void init() {
    int u,v,w;
    scanf("%d%d%d", &n, &m, &mm);
    s = 1;
    for(int i = 1;i <= n;i++) dist[i] = INF;
    for(int i = 0;i < m;i++) {
        scanf("%d%d%d", &u, &v, &w);
        edge[i].from = u;
        edge[i].to = v;
        edge[i].w = w;
        edge[i + m].from = v;
        edge[i + m].to = u;
        edge[i + m].w = w;
    }
    dist[s] = 0;
    for(int i = m;i <m + mm;i++){
        scanf("%d%d%d", &u, &v, &w);
        edge[i].from = u;
        edge[i].to = v;
        edge[i].w = -w;
    }
    m+=mm;
    return;
}
bool Bellman_Ford() {
    int u,v,w,flag;
    for(int i = 1;i <= n;i++) {
        flag = 0;
        for(int j = 0;j < m;j++) {
            u = edge[j].from;
            v = edge[j].to;
            w = edge[j].w;
            if(dist[v] > dist[u] + w) {
                dist[v] = dist[u] + w;
                flag = 1;
            }
        }
        if(!flag) break;
        if(i == n && flag) return false;
    }
    return true;
}
```

```cpp
// 邻接矩阵

#include <cstdio>
#include <string>
#include <string.h>
const int M = 205;
const int INF = 1 << 28;
int n; // 点数
int m;  // 边数
int d[M][M];    // 记录边权
void init();
void Floyd();
int main(){
    while(scanf("%d%d", &n, &m) != EOF){
        init();
        Floyd();
    }
    return 0;
}
void init(){
    int u,v,w;
    for(int i = 1;i <= n;i++){
        for(int j = 1;j <= n;j++){
            d[i][j] = i == j ? 0 : INF;
        }
    }
    for(int i = 0;i < m;i++){
        scanf("%d%d%d", &u, &v, &w);
        if(d[u][v] > w) d[u][v] = w;
        d[v][u] = d[u][v];
    }
    return;

void Floyd(){
    for(int k = 1;k <= n;k++){
        for(int i = 1;i <= n;i++){
            for(int j = 1;j <= n;j++){
                if(d[i][j] > d[i][k] + d[k][j]){
                    d[i][j] = d[i][k] + d[k][j];
                }
            }
        }
    }
    return;
}
```

## Prim(O(nlogn))
```cpp
/*
求无负权有向图的最短路径的Dijkstra算法， 用二叉堆实现
时间复杂度O(mlogn)
*/
#include <cstdio>
#include <list>
using namespace std;
const int mn = 1001, inf = 1 << 30;
int n, m, s, res, cnt, hs, d[mn], p[mn], h[mn], pos[mn];
struct lnode{ int v, w;};
list<lnode> g[mn];
inline void swaph(int a, int b) {
    swap(h[a], h[b]);
    pos[h[a]] = a;
    pos[h[b]] = b;
}

void init() {
    scanf("%d%d", &n, &m);
    res = 0, cnt = 0;
    for (int i = 1;i <= n; i++) {
        g[i].clear();
        d[i] = inf;
        h[i] = pos[i] = i;
        p[i] = 0;
    }
    d[s] = 0; 
    hs = n;
    s = 1;
    swaph(1, s);
    for (int i = 1, a,b,c; i <= m; i++) {
        scanf("%d%d%d", &a, &b, &c);
        g[a].push_back({b, c});
        // g[b].push_back((lnode){a, c}); 加上此句为无向图
    }
    return;
}

void shiftup(int p) {
    for (int i = p; i > 1 && d[h[i]] < d[h[i/2]]; i /= 2) {
        swaph(i, i/2);
    }
    return;
}

int delmin() {
    int ret = h[1];
    swaph(1, hs--);
    for (int i =1, tmp; i <= hs/2;i = tmp) {
        tmp = i * 2;
        if (tmp < hs && d[h[tmp+1]] < d[h[tmp]]) tmp++;
        if (d[h[i]] <= d[h[tmp]]) break;
        swaph(i, tmp);
    }
    return ret;
}

void prim() {
    while(hs) {
        int mp = delmin();
        if (d[mp] < inf) {
            res += d[mp];
            cnt++;
        }
        d[mp] = 0;
        for (list<lnode>::iterator it = g[mp].begin(); it != g[mp].end(); it++) {
            if (pos[it->v] <= hs && d[it->v] > d[mp] + it->w) {
                d[it->v] = d[mp] + it->w;
                p[it->v] = mp;
                shiftup(pos[it->v]);
            }
        }
    }
    return;
}

void output() {
    if (cnt < n) {
        printf("No way!");
        return;
    }
    printf("Distance is %d\n", res);
}

int main() {
    init();
    prim();
    output();
    return 0;
}
```

## Kruskal O(E * logE)
```cpp
// 注意边数上限不能太小

#include <cstdio>
#include <string>
#include <algorithm>

using namespace std;

const int M = 1005;
const int N = 10005;

struct node {
    int from;
    int to;
    int w;
}edge[N];  // from 起点，to 终点，w 权值

int n, m, res;
int fa[M];
int rk[M];

void INIT();
void sol();
int Find(int k);
void Union(int x, int y);
bool Kruskal();

bool cmp(const node &a, const node &b) {
    return a.w < b.w;
}

int main() {
    while (scanf("%d%d", &n, &m) != EOF) {
        INIT();
        sol();
    }
    return 0;
}

void INIT() {
    int u, v, w;
    for (int i = 0;i < m; i++) {
        scanf("%d%d%d", &u, &v, &w);
        edge[i].from = u;
        edge[i].to = v;
        edge[i].w = w;
    }
    for (int i = 1;i <= n; i++) {
        fa[i] = i;
        rk[i] = 0;
    }
    res = 0;
    m *= 2;
    sort(edge, edge + m, cmp);
    return;
}

void sol() {
    if(Kruskal()) {
        printf("%d\n", res);
    }else {
        printf("Fail to construct a connected map.\n");
    }
    return;
}

int Find(int x) {
    if (x == fa[x]) return x;
    else return fa[x] = Find(fa[x]);
}

void Union(int x, int y) {
    x = Find(x);
    y = Find(y);
    if (x != y) {
        if (rk[x] < rk[y]) fa[x] = y;
        else if (rk[x] > rk[y]) fa[y] = x;
        else {
            fa[x] = y;
            rk[y]++;
        }
    }
    return;
}

bool Kruskal() {
    for (int i = 0;i < m; i++) {
        if (Find(edge[i].from) != Find(edge[i].to)) {
            Union(edge[i].from, edge[i].to);
            res += edge[i].w;
        }
    }
    for (int i = 2;i <= n; i++) {
        if (Find(i) != Find(1)) return false;
    }
    return true;
}
```

## 3.10 缩点
```cpp
// 缩点，两遍DFS
#include <cstdio>
#include <string>
#include <string.h>
#include <vector>

using namespace std;

const int M = 10005;
const int INF = 1 << 30;
int n, m; // n 点数，m 边数
int top; // 记录队列
int mark; // mark记录缩点之后的点
vector<int> Map[M]; // 记录原图
vector<int> un_Map[M]; // 记录反向图
vector<int> daMap[M]; // 记录sec图
int cover[M]; // 记录每个点是否被覆盖
int queue[M]; // 记录反向图遍历的顺序
int belong[M]; // 记录缩点之后的新点

void INIT();
void sol();
void sec();// 用于两遍DFS, 缩点
void compress(); // 用于压缩点，求sec图
void DFS(int k); 
void DFS_2(int k);

int main() {
    while (scanf("%d%d", &n, &m) != EOF) {
        if (n == 0 && m == 0) break;
        INIT();
        sol();
    }
    return 0;
}

void INIT() {
    int u, v;
    top = 0; mark = 0;
    for (int i = 1;i <= n; i++) {
        Map[i].clear();
        un_Map[i].clear();
        daMap[i].clear();
    }
    for (int i = 0;i < m; i++) {
        scanf("%d%d", &u, &v);
        Map[u].push_back(v);
        un_Map[v].push_back(u);
    }
    return;
}

void sol() {
    sec();
    compress();
    return;
}

void sec() {
    memset(cover, 0, sizeof(cover));
    for (int i = 1;i <= n; i++) {
        if (!cover[i]) DFS(i);
    }
    memset(cover, 0, sizeof(cover));
    for (int i = top;i > 0; i--) {
        if (!cover[queue[i]]) {
            mark++;
            DFS_2(queue[i]);
        }
    }
    return;
}

void compress() {
    for (int i = 1;i <= n; i++) {
        for (int j = 0;j < Map[i].size(); j++) {
            if (belong[i] != belong[Map[i][j]]) {
                daMap[belong[i]].push_back(belong[Map[i][j]]);
            }
        }
    }
    return;
}

void DFS(int k) {
    cover[k] = 1;
    for (int i = 0;i < Map[k].size(); i++) {
        if (!cover[Map[k][i]]) DFS(Map[k][i]);
    }
    cover[k] = 2;
    queue[++top] = k;
    return;
```

## 3.11 欧拉路（链表）
```cpp
// 无向图: 所有节点度数都为偶数；度数为奇数的节点数只有两个
// 有向图: 所有节点出度等于入度；出度比入度大1的点只有一个（该点为起点），入度比出度大1的点只有一个（该点为终点）
// 求欧拉路: 找一个环 p, v1, v2, ... p， 再枚举每一个点，递归地找环；
// 解法: 一遍DFS
#include <cstdio>
#include <string>
#include <string.h>
#include <algorithm>

using namespace std;

const int M = 1005;
const int N = 100005;
struct node {
    int from;
    int to;
} edge[N];
node ans[N];
struct node2 {
    int id;
    node2 *next;
}d[M]; // 记录每个点的出边
node2 pool[N]; // 内存池
int n, m, cnt, flag;
int s, t;
int degi[M]; // 记录入度
int dego[M]; // 记录出度
int cover[N]; // 记录每个边是否访问过

void INIT();
void sol();
void DFS(int u);
void Output();
bool cmp(const node &a, const node &b) {return a.from < b.from;}

int main() {
    int t;
    scanf("%d", &t);
    while (t--) {
        INIT();
        sol();
        Output();
    }
    return 0;
}

void INIT() {
    int u, v, top;
    memset(degi, 0, sizeof(degi));
    memset(dego, 0, sizeof(dego));
    memset(cover, 0, sizeof(cover));
    scanf("%d %d", &n, &m);
    for (int i = 1; i <= n;i++) d[i].next = NULL;
    top = 0; flag = 0; s = 0; t = 0; cnt = 0;
    for (int i = 0;i < m;i++) {
        scanf("%d %d", &u, &v);
        dego[u]++; degi[v]++;
        edge[i].from = u;
        edge[i].to = v;
    }
    sort(edge, edge + m, cmp);
    for (int i = 0;i < m;i++){
        u = edge[i].from;
        pool[top].id = i;
        pool[top].next = d[u].next;
        d[u].next = &pool[top++];
    }
    for (int i = 1;i <= n; i++) {
        if (degi[i] != dego[i]) {
            if (dego[i] - degi[i] == 1 && s == 0) s = i;
            else if (degi[i] - dego[i] == 1 && t == 0) t = i;
            else {flag = 1; break;}
        }
    }
    if (s == 0) s = edge[0].from;
    return;
}

void sol() {
    if (flag == 1) return;
    if (s == 0) s = edge[0].from;
    DFS(s);
    return;
}

void DFS(int u) {
    int id, v;
    node2 *s;
    s = d[u].next;
    while (s != NULL) {
        id = s->id;
        if (cover[id] == 0) {
            cover[id] = 1;
            v = edge[id].to;
            DFS(v);
            ans[cnt++] = edge[id];
        }
        s = s->next;
    }
    return;
}

void Output() {
    if (cnt < m) flag = 1;
    if (flag == 1) printf("No solution\n");
    else {
        for (int i = cnt - 1;i >= 0;i--) {
            printf("%d %d\n", ans[i].from, ans[i].to);
        }
    }
    return;
}
```

## 3.14 Difference Constraint System(差分约束系统)
```cpp
/* 差分约束系统
将不等式转化成图来求解
1. 如果要求某个值的最大值，在建图的时候，就要把所有的不等式转化成这样的方程:
 A - B <= C 之后添加边: B -> A, 权值为 C。 dist初始化为正无穷，求最短路
2. 如果要求某个值的最小值，在建图的时候，就要把所有的不等式转化成这样的方程:
    A - B >= C 之后添加边: A -> B, 权值为 C。 dist初始化为负无穷，求最长路
有负环说明无解，否则解为dist数组的值
*/
#include <cstdio>
#include <string>

const int M = 50005;
const int INF = 100000005;

struct node {
    int u, v, c;
} edge[M *4];

int dist[M];
bool Bellman(int s, int n, int m);

int main() {
    int n, m, s, res;
    while(scanf("%d", &m) != EOF) {
        n = 0;
        for (int i = 0;i < m; i++) {
            scanf("%d %d %d", &edge[i].u, &edge[i].v, &edge[i].c);
            edge[i].v = edge[i].u - 1;
            edge[i].c = -edge[i].c;
            n = n > edge[i].u ? n : edge[i].u;
        }
        s = n + 1;
        for (int i = m; i < m + n; i++) {
            edge[i].v = i - m;
            edge[i].u = i - m + 1;
            edge[i].c = 0;
            edge[i+n].v = i - m + 1;
            edge[i+n].u = i - m;
            edge[i+n].c = 1;
        }
        for (int i = m + 2 * n; i <= m + 2 * n + n; i++) {
            edge[i].u = s;
            edge[i].v = i - (m + 2 * n);
            edge[i].c = 0;
        }
        if (Bellman(s, n + 2, m + 3 * n  + 1)) {
            res = dist[n];
            if (dist[0] < 0) res -= dist[0];
            printf("%d\n", res);
        }
    }
    return 0;
}

bool Bellman(int s, int n, int m) {
    int u, v, c, flag;
    for(int i = 0;i < n;i++) dist[i] = INF;
    dist[s] = 0;
    for(int i = 1;i <= n;i++) {
        flag = 0;
        for (int j = 0;j < m;j++) {
            u = edge[j].u;
            v = edge[j].v;
            c = edge[j].c;
            if (dist[u] + c < dist[v]) {
                dist[v] = dist[u] + c;
                flag = 1;
            }
        }
        if (flag == 0) break;
        if (i == n && flag == 1) return false;
    }
    return true;
}
```

## 3.15 拓扑序
```cpp
#include <cstdio>

const int M = 30;

int d[M][M];
int queue[M];
int dei[M];
int main() {
    int t, n, m, top, u, v, p;
    scanf("%d", &t);
    while (t--) {
        scanf("%d %d", &n, &m);
        top = 0;
        memset(d, 0, sizeof(d));
        memset(dei, 0, sizeof(dei));
        for (int i = 0;i < m;i++) {
            scanf("%d %d", &u, &v);
            d[u][v] = 1;
            dei[v]++;
        }

        // 每次取入度为0的点入队，队中P弹出，所有相连通的点入度-1, 重复操作
        for (int i = 1;i <= n;i++) {
            if (dei[i] == 0) queue[top++] = i;
        }
        for (int i = 0;i < top;i++) {
            p = queue[i];
            for (int j = 1;j <= n;j++) {
                if (d[p][j] == 1) {
                    d[p][j] = 0;
                    dei[j]--;
                    if (dei[j] == 0) queue[top++] = j;
                }
            }
        }
        if (top == n) printf("1\n");
        else printf("0\n");
    }
    return 0;
}
```
