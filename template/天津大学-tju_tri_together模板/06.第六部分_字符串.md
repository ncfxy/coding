# 第六部分 字符串

## 6.1 KMP算法 O(n+m)
```cpp
#include <cstdio>
#include <string>
#include <iostream>
using namespace std;

const int M = 10005;

int n, m;
int nextArray[M];
char P[M]; // 母串
char T[M];

void Next(const char[]);
bool Mat(const char [], const char []);

int main() {
    while(scanf("%s%s", P, T) != EOF) {
        Next(T);
        if (Mat(P, T)) {
            printf("YES\n");
        } else {
            printf("NO\n");
        }
    }
    return 0;
}

void Next(const char T[]) {
    memset(nextArray, 0, sizeof(nextArray));
    int j, t;
    t = nextArray[0] = -1;
    j = 0;
    m = strlen(T);
    while(j < m -1) {
        if (t < 0 || T[j] == T[t]) nextArray[++j] = ++t;
        else t = nextArray[t];
    }
    return;
}

bool Mat(const char P[], const char T[]) {
    bool r;
    int i,j;
    n = strlen(P);
    i = j = 0;
    while(i < n && j < m) {
        if (j < 0 || P[i] == T[j]) {
            i++;
            j++;
        } else j = nextArray[j];
    }
    if (j == m) r = true;
    else r = false;
    return r;
}
```

## 6.2 扩展KMP
```cpp
#include <cstdio>
#include <string>
using namespace std;

const int M = 1005;

int n, m;
int extend[M];
int nextArr[M];
char P[M], T[M]; // P为母串，T为子串

void Next(const char[]);
void GetExtend(const char[], const char[]);

int main() {
  while (scanf("%s%s", P, T) != EOF) {
    Next(T);
    GetExtend(P, T);
    for (int i = 0; i < n; i++)
      printf("%d ", extend[i]);
    printf("\n");
    for (int j = 0; j < m; j++)
      printf("%d ", nextArr[j]);
    printf("\n");
  }
  return 0;
}

void Next(const char T[]) {
    int j, t, a, p;
    t = -1;
    m = strlen(T); 
    a = p = 0;
    memset(nextArr, 0, sizeof(nextArr));
    nextArr[0] = m;
    for(j = 1;j < m;++j, --t) {
        if (t < 0 || j + nextArr[j-a] >= p) {
            if (t < 0) {
                t = 0; p = j;
            }
            while(p < m && t < m && T[p] == T[t]) {
                ++p;
                ++t;
            }
            nextArr[j] = t;
            a = j;
        } else {
            nextArr[j] = nextArr[j-a];
        }
    }
    return;
}

void GetExtend(const char P[], const char T[]) {
    int i, j, a, p;
    j = -1;
    memset(extend, 0, sizeof(extend));
    n = strlen(P);
    for(i = 0;i < n;++i, --j) {
        if (j < 0 || i + nextArr[i-a] >= p) {
            if(j < 0) {
                j = 0;
                p = i;
            }
            while(p < n && j < m && P[p] == T[j]) {
                ++p;
                ++j;
            }
            extend[i] = j;
            a = i;
        } else {
            extend[i] = nextArr[i-a];
        }
    }
    return;
}
```

## 6.3 Rabin_Karp
```cpp
#include <cstdio>
#include <iostream>
#include <string>
using namespace std;

const int M = 16000000;
const int q = 6999997;
const int MAX_N = 150;
int n, m, h, d;
char S[M];
char T[M];
int Rabin_Karp();
int cover[MAX_N];
int main() {
  int pos;
  while (scanf("%s%s", S, T) != EOF) {
    memset(cover, 0, sizeof(cover));
    d = 0;
    for (int i = 0; i < strlen(S); i++)
      if (!cover[S[i]]) {
        cover[S[i]] = 1;
        d++;
      }
    for (int j = 0; j < strlen(T); j++) {
      if (!cover[T[j]]) {
        cover[T[j]] = 1;
        d++;
      }
    }
    pos = Rabin_Karp();
    printf("%d\n", pos);
  }
  return 0;
}

int Rabin_Karp() {
  n = strlen(S);
  m = strlen(T);
  if (n < m)
    return -1; // T 不可能是S的子串
  int p, t;
  h = 1;
  for (int i = 0; i < m; i++)
    h = h * d % q; // 构造h (d^m)
  p = t = 0;       // 计算 p 和 t0
  for (int i = 0; i < m; i++) {
    p = (d * p + S[i]) % q;
    t = (d * t + T[i]) % q;
  }
  for (int i = 0; i <= n - m; i++) {
    if (i != 0) {
      p = (d * p - (S[i - 1]) * h + S[i - 1 + m]) % q;
      if (p < 0)
        p += q; // 求下一个位置的P值
    }
    if (p == t) {
      int j;
      for (j = 0; j < m; j++)
        if (T[j] != S[i + j])
          break;
      if (j == m)
        return i;  // 为预防伪命中，重新匹配一遍
    }
  }
  return -1;
}
```

## 6.4 AC自动机
```cpp
// AC自动机
// m 个单词，总长 L
// 1 个正文，总长 n
// 搜索正文中有多少单词库里的单词(个数，而非次数，如果求次数，不加 tmp->cnt =
// -1;) O(m*Li + n) hdu 2222 AC自动机份三步:
// 1. 构造一棵Trie树
// 2. 构造失败指针
// 3. 模式匹配过程
#include <cstdio>
#include <string>
using namespace std;

const int M = 500010;
const int N = 1000010;
const int MAX_N = 26;

struct TrieNode {
  TrieNode *next;
  TrieNode *son[MAX_N];
  bool end;
  int cnt;

  TrieNode() {
    next = NULL;
    memset(son, 0, sizeof(son));
    end = false;
    cnt = 0;
  }
} *queue[M], *root;

int n;
char str[N];
void Init();
void TrieInsert(char *);
void Build_AC_Automachine();
int AC_Search();

int main() {
  int t;
  scanf("%d", &t);
  while (t--) {
    Init();
    Build_AC_Automachine();

    scanf("%s", str);
    printf("%d\n", AC_Search());
  }
  return 0;
}

void Init() {
  char buf[100];
  root = new TrieNode();
  scanf("%d", &n);
  for (int i = 0; i < n; i++) {
    scanf("%s", buf);
    TrieInsert(buf);
  }
  return;
}

void TrieInsert(char *buf) {
  int v, pts;
  pts = 0;
  // 开始时指针指向 root
  TrieNode *ptt = root;
  while (buf[pts]) {
    v = buf[pts] - 'a';
    // ptt 一直指向当前节点的父亲，如果没有这个儿子，新建节点
    if (ptt->son[v] == NULL)
      ptt->son[v] = new TrieNode();
    // ptt 指向这个 “新儿子节点”
    ptt = ptt->son[v];
    // pts 继续后移
    pts++;
  }
  // 单词末尾标记
  ptt->end = true;
  // 数目++
  ptt->cnt++;
  return;
}

void Build_AC_Automachine() {
  int top = 0;
  queue[top++] = root;
  // root->next 指向空
  root->next = NULL;
  for (int i = 0; i < top; i++) {
    TrieNode *cur = queue[i];
    for (int j = 0; j < MAX_N; j++) {
      if (cur->son[j] != NULL) {
        // 根节点的所有儿子 ->next 指向根节点
        if (cur == root)
          cur->son[j]->next = root;
        // cur 指向当前节点的父亲，ptt指向父节点的前向指针
        else {
          TrieNode *ptt = cur->next;
          while (ptt != NULL) {
            // 如果父节点中有这个儿子
            if (ptt->son[j] != NULL) {
              // 把当前节点(cur->son[j])的next指针指向这个儿子节点
              cur->son[j]->next = ptt->son[j];
              // 如果这个节点是单词结尾，标记前驱
              if (ptt->son[j]->end == true)
                cur->son[j]->end = true;
              break;
            }
            // 如果父节点没有这个儿子沿着它父节点(ptt)的next回溯，直到指向root
            ptt = ptt->next;
          }
          // 如果没有成功找到next, 指向root
          if (ptt == NULL)
            cur->son[j]->next = root;
        }
        // 把这个儿子节点加入队列
        queue[top++] = cur->son[j];
      }
    }
  }
  return;
}

int AC_Search() {
    int pts, ans;
    // 新建ptt指向root, pts 指向字符串首位置
    pts = ans = 0;
    TrieNode *ptt = root;
    while(str[pts]) {
        int v = str[pts] - 'a';
        // ptt 指向当前节点，如果没有这个儿子
        while(ptt->son[v] == NULL && ptt != root) ptt = ptt->next;

        // ptt 指向当前节点的儿子节点
        // 此处先指向儿子，再讨论是否有点满足ppt 指向讨论节点父亲的原则
        ptt = ptt->son[v];

        // 如果还是没有该节点(c)，指向根节点（因为已经没有可以回溯的点了）
        if(ptt == NULL) ptt = root;

        // 如果此处有该节点(c)
        // 如果ptt过了一个cnt!=-1 && cnt > 0的点，就说明该单词出现过了 ans += cnt
        // 如果ptt回溯到了一个 cnt != -1 && cnt > 0 的点，就说明该单词已经出现过了 ans += cnt
        // 一旦到 root, 终止循环； 一旦cnt = -1,即已经访问过了，以前的回溯都访问过了
        // 我的理解: 每找到一个点，就把其所有回溯都加了
        TrieNode *tmp = ptt;
        // 计算个数
        while(tmp != NULL && tmp->cnt != -1) {
            ans += tmp->cnt;
            tmp->cnt = -1;
            tmp = tmp->next;
        }
        // 计算次数
        while(tmp != NULL) {
            ans += tmp->cnt;
            tmp = tmp->next;
        }
        pts++;
    }
    return ans;
}
```

## DC3 Suffix(后缀数组)
```cpp
#include <cstdio>
#include <string>
using namespace std;

inline bool leq(int a1, int a2, int b1, int b2) { // lexic, orders for pairs
  return (a1 < b1 || a1 == b1 && a2 <= b2);
}

inline bool leq(int a1, int a2, int a3, int b1, int b2, int b3) {
  return (a1 < b1 || a1 == b1 && leq(a2, a3, b2, b3));
}
static void radixPass(int *a, int *b, int *r, int n,
                      int K) { // count occurrences
  int *c = new int[K + 1];
  for (int i = 0; i <= K; i++)
    c[i] = 0;
  for (int i = 0; i < n; i++)
    c[r[a[i]]]++;
  for (int i = 0, sum = 0; i <= K; i++) {
    int t = c[i];
    c[i] = sum;
    sum += t;
  }
  for (int i = 0; i < n; i++)
    b[c[r[a[i]]]++] = a[i];
  delete[] c;
}

// find the suffix array SA of s[0...n-1] in {1...K}^n
// require s[n] = s[n+1] = s[n+2] = 0, n>= 2
void suffixArray(int *s, int *SA, int n, int K) {
  int n0 = (n + 2) / 3, n1 = (n + 1) / 3, n2 = n / 3, n02 = n0 + n2;
  int *s12 = new int[n02 + 3];
  s12[n02] = s12[n02 + 1] = s12[n02 + 2] = 0;
  int *SA12 = new int[n02 + 3];
  SA12[n02] = SA12[n02 + 1] = SA12[n02 + 2] = 0;
  int *s0 = new int[n0];
  int *SA0 = new int[n0];

  // generate positions of mod 1 and mod 2 suffixes
  // the "+(n0-n1)" adds a dummy mod 1 suffix if n % 3 == 1
  for (int i = 0, j = 0; i < n + (n0 - n1); i++) {
    if (i % 3 != 0)
      s12[j++] = i;
  }

  // lsb radix sort the mod 1 and mod 2 triples
  radixPass(s12, SA12, s + 2, n02, K);
  radixPass(SA12, SA0, s + 1, n02, K);
  radixPass(s12, SA12, s, n02, K);
  // find lexicographic names of triples
  int name = 0, c0 = -1, c1 = -1, c2 = -1;
  for (int i = 0; i < n02; i++) {
    if (s[SA12[i]] != c0 || s[SA12[i] + 1] != c1 || s[SA12[i] + 2] != c2) {
      name++;
      c0 = s[SA12[i]];
      c1 = s[SA12[i] + 1];
      c2 = s[SA12[i] + 2];
    }
    if (SA12[i] % 3 == 1) {
      s12[SA12[i] / 3] = name; // left half
    } else {
      s12[SA12[i] / 3 + n0] = name; // right half
    }
  }
  // recurse if names are not yet unique
  if (name < n02) {
    suffixArray(s12, SA12, n02, name);
    // store unique names in s12 using the suffix array
    for (int i = 0; i < n02; i++)
      s12[SA12[i]] = i + 1;
  } else {
    // generate the suffix array of s12 directly
    for (int i = 0; i < n02; i++)
      SA12[s12[i] - 1] = i;
  }

  // stably sort the mod 0 suffixes from SA12 by their first character
  for (int i = 0, j = 0; i < n02; i++)
    if (SA12[i] < n0)
      s0[j++] = 3 * SA12[i];
  radixPass(s0, SA0, s, n0, K);
  // merge sorted SA0 and SA12
  for (int p = 0, t = n0 - n1, k = 0; k < n; k++) {
#define GetI() (SA12[t] < n0 ? SA12[t] * 3 + 1 : (SA12[t] - n0) * 3 + 2)
    int i = GetI();
    int j = SA0[p];
    if (SA12[t] < n0 ? leq(s[i], s12[SA12[t] + n0], s[j], s12[j / 3])
                     : leq(s[i], s[i + 1], s12[SA12[t] - n0 + 1], s[j],
                           s[j + 1], s12[j / 3 + n0])) {
      SA[k] = i;
      t++;
      if (t == n02) {
        for (k++; p < n0; p++, k++)
          SA[k] = SA0[p];
      }
    } else {
      SA[k] = j;
      p++;
      if (p == n0) { // done -- only SA12 suffixes left
        for (k++; t < n02; t++, k++)
          SA[k] = GetI();
      }
    }
  }
  delete[] s12;
  delete[] SA12;
  delete[] s0;
  delete[] SA0;
  return;
}

const int MAXLEN = 200010;
char buf1[MAXLEN], buf2[MAXLEN];
int a[MAXLEN], sa[MAXLEN], ranks[MAXLEN], height[MAXLEN], h[MAXLEN];

void calc_height(int n) {
    int i,s,t,k;
    for(i = 0;i < n;i++){
        if (ranks[i] == 0) h[i] = 0;
        else if(i == 0 || h[i-1] <=1) {
            for(k =0;a[i+k] ==a[sa[ranks[i]-1]+k];k++);
            h[i] = k;
        } else {
            for(k = h[i-1] -1;a[i+k] == a[sa[ranks[i]-1]+k];k++);
            h[i] = k;
        }
    }
    return;
}

int main() {
    int k, len;
    while(scanf("%s", &buf1) != EOF) {
        // a means cover
        memset(a, 0, sizeof(a));
        len = strlen(buf1);
        k = 26;
        for(int i = 0;i < len;i++)a[i] = buf1[i] - 'a' + 1;
        suffixArray(a, sa, len, k);
        for(int i =0;i < len;i++) printf("%d", sa[i]);
        printf("\n");
        for(int i =0;i < len;i++) ranks[sa[i]] = i;
        calc_height(len);
        for(int i = 0;i < len;i++) height[ranks[i]] = h[i];
        for(int i = 0;i < len;i++) printf("%d", height[i]);
        printf("\n");
    }
    return 0;
}
```
