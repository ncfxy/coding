## 1.2 Prim算法（稀疏图）
```cpp
#include <bits/stdc++.h>
using namespace std;

using ll = long long;

const int MAXN = 200005;
const ll INF = (1LL << 60);

struct Edge {
    int to;
    ll w;
};

vector<Edge> g[MAXN];
bool vis[MAXN];

// 返回最小生成树的边权和
// 如果图不连通，返回 -1
ll prim(int n) {
    fill(vis + 1, vis + n + 1, false);

    // {边权, 节点}
    priority_queue<
        pair<ll, int>,
        vector<pair<ll, int>>,
        greater<pair<ll, int>>
    > pq;

    // 从 1 号节点开始
    pq.push({0, 1});

    ll ans = 0;
    int cnt = 0;

    while (!pq.empty()) {
        auto [w, u] = pq.top();
        pq.pop();

        if (vis[u]) continue;

        // 把 u 加入最小生成树
        vis[u] = true;
        ans += w;
        cnt++;

        // 把 u 能扩展到的边加入优先队列
        for (auto [v, cost] : g[u]) {
            if (!vis[v]) {
                pq.push({cost, v});
            }
        }
    }

    // 没有访问到所有节点，说明图不连通
    if (cnt != n) {
        return -1;
    }

    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;
    cin >> n >> m;

    for (int i = 0; i < m; i++) {
        int u, v;
        ll w;
        cin >> u >> v >> w;

        // 最小生成树针对无向图
        g[u].push_back({v, w});
        g[v].push_back({u, w});
    }

    ll ans = prim(n);

    if (ans == -1) {
        cout << "Impossible\n";
    } else {
        cout << ans << '\n';
    }

    return 0;
}
```

## 1.3 Prim算法（稠密图）（领接矩阵）
- n <= 1000
- 图非常稠密，那么可以直接使用 \(O(n^2)\) Prim，代码反而更简单
```cpp
const ll INF = (1LL << 60);

ll g[MAXN][MAXN];
ll dist[MAXN];
bool vis[MAXN];

ll prim(int n) {
    fill(dist + 1, dist + n + 1, INF);
    fill(vis + 1, vis + n + 1, false);

    dist[1] = 0;

    ll ans = 0;

    for (int i = 1; i <= n; i++) {
        int u = -1;

        // 找到生成树外 dist 最小的节点
        for (int j = 1; j <= n; j++) {
            if (!vis[j] &&
                (u == -1 || dist[j] < dist[u])) {
                u = j;
            }
        }

        // 无法继续扩展，图不连通
        if (u == -1 || dist[u] == INF) {
            return -1;
        }

        vis[u] = true;
        ans += dist[u];

        // 用 u 更新其它节点
        for (int v = 1; v <= n; v++) {
            if (!vis[v] && g[u][v] < dist[v]) {
                dist[v] = g[u][v];
            }
        }
    }

    return ans;
}
```

## 1.4 Kruskal算法
- 边按权值排序 + 并查集判断是否成环
```cpp
#include <bits/stdc++.h>
using namespace std;

using ll = long long;

const int MAXN = 200005;

struct Edge {
    int u, v;
    ll w;

    bool operator < (const Edge& other) const {
        return w < other.w;
    }
};

int fa[MAXN];
int sz[MAXN];

int find(int x) {
    if (fa[x] == x) return x;
    return fa[x] = find(fa[x]);
}

bool unite(int x, int y) {
    x = find(x);
    y = find(y);

    if (x == y) return false;

    // 按集合大小合并
    if (sz[x] < sz[y]) {
        swap(x, y);
    }

    fa[y] = x;
    sz[x] += sz[y];

    return true;
}

ll kruskal(int n, vector<Edge>& edges) {
    // 初始化并查集
    for (int i = 1; i <= n; i++) {
        fa[i] = i;
        sz[i] = 1;
    }

    // 边按权值从小到大排序
    sort(edges.begin(), edges.end());

    ll ans = 0;
    int cnt = 0;

    for (auto [u, v, w] : edges) {
        // u 和 v 不在同一个连通块中
        if (unite(u, v)) {
            ans += w;
            cnt++;

            // 最小生成树只需要 n - 1 条边
            if (cnt == n - 1) {
                break;
            }
        }
    }

    // 不足 n - 1 条边，说明图不连通
    if (cnt != n - 1) {
        return -1;
    }

    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;
    cin >> n >> m;

    vector<Edge> edges;

    for (int i = 0; i < m; i++) {
        int u, v;
        ll w;

        cin >> u >> v >> w;

        edges.push_back({u, v, w});
    }

    ll ans = kruskal(n, edges);

    if (ans == -1) {
        cout << "Impossible\n";
    } else {
        cout << ans << '\n';
    }

    return 0;
}
```

## 1.5 矩阵树定理（Matrix-Tree Theorem）
- 计算一个无向图的生成树数量
```cpp
#include <bits/stdc++.h>
using namespace std;

using ll = long long;

const int MAXN = 505;

ll a[MAXN][MAXN];
ll MOD;

ll qpow(ll a, ll b) {
    ll res = 1;

    while (b) {
        if (b & 1) {
            res = res * a % MOD;
        }

        a = a * a % MOD;
        b >>= 1;
    }

    return res;
}

// 计算 n * n 矩阵的行列式 mod MOD
// 要求 MOD 为质数
ll determinant(int n) {
    ll ans = 1;

    for (int i = 0; i < n; i++) {
        int pivot = -1;

        for (int j = i; j < n; j++) {
            if (a[j][i] % MOD != 0) {
                pivot = j;
                break;
            }
        }

        // 这一列全是 0
        if (pivot == -1) {
            return 0;
        }

        // 交换行
        if (pivot != i) {
            for (int j = i; j < n; j++) {
                swap(a[i][j], a[pivot][j]);
            }

            ans = (MOD - ans) % MOD;
        }

        ll x = (a[i][i] % MOD + MOD) % MOD;

        ans = ans * x % MOD;

        ll inv = qpow(x, MOD - 2);

        // 消掉下面所有元素
        for (int j = i + 1; j < n; j++) {
            ll factor = (a[j][i] % MOD + MOD) % MOD;
            factor = factor * inv % MOD;

            for (int k = i; k < n; k++) {
                a[j][k] =
                    (a[j][k] - factor * a[i][k]) % MOD;

                if (a[j][k] < 0) {
                    a[j][k] += MOD;
                }
            }
        }
    }

    return ans % MOD;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;
    cin >> n >> m >> MOD;

    // Laplacian 矩阵
    vector<vector<ll>> L(n, vector<ll>(n, 0));

    for (int i = 0; i < m; i++) {
        int u, v;
        cin >> u >> v;

        --u;
        --v;

        // 无向边 u-v
        L[u][u]++;
        L[v][v]++;

        L[u][v]--;
        L[v][u]--;
    }

    // 删除第 n-1 行、第 n-1 列
    // 将前 n-1 行列复制到 a
    for (int i = 0; i < n - 1; i++) {
        for (int j = 0; j < n - 1; j++) {
            a[i][j] = (L[i][j] % MOD + MOD) % MOD;
        }
    }

    cout << determinant(n - 1) << '\n';

    return 0;
}
```
