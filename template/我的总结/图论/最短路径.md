# 最短路径算法汇总

## 1.1 各算法对比

| 算法 | 负权边 | 检测负环 | 时间复杂度 |
|---|---|---|---|
| Dijkstra | ❌ | ❌ | \(O(m\log n)\) |
| Bellman-Ford | ✅ | ✅ | \(O(nm)\) |
| SPFA | ✅ | ✅ | 最坏 \(O(nm)\) |

竞赛里可以记一句：**无负边优先 Dijkstra，有负边考虑 Bellman-Ford / SPFA；要求稳定复杂度时优先 Bellman-Ford。**

## 1.2 Dijkstra

```cpp
#include <iostream>
#include <vector>
#include <queue>
using namespace std;

using ll = long long;

const ll INF = (1LL << 60);
const int MAXN = 200005;

struct Edge {
    int to;
    ll w;
};

vector<Edge> g[MAXN];
ll dist[MAXN];
bool vis[MAXN];

void dijkstra(int s, int n) {
    // 初始化
    for (int i = 1; i <= n; i++) {
        dist[i] = INF;
        vis[i] = false;
    }

    // 小根堆：{当前距离, 节点}
    priority_queue<
        pair<ll, int>,
        vector<pair<ll, int>>,
        greater<pair<ll, int>>
    > pq;

    dist[s] = 0;
    pq.push({0, s});

    while (!pq.empty()) {
        // auto [d, u] = pq.top();
        auto d = pq.top().first;
        auto u = pq.top().second;
        pq.pop();

        // 这个点已经确定过最短距离
        if (vis[u]) continue;
        vis[u] = true;

        for (auto edge : g[u]) {
            int v = edge.to;
            ll w = edge.w;
            if (dist[v] > dist[u] + w) {
                dist[v] = dist[u] + w;
                pq.push({dist[v], v});
            }
        }
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m, s;
    cin >> n >> m >> s;

    for (int i = 0; i < m; i++) {
        int u, v;
        ll w;
        cin >> u >> v >> w;

        // 有向图
        g[u].push_back({v, w});

        // 如果是无向图，再加：
        // g[v].push_back({u, w});
    }

    dijkstra(s, n);

    for (int i = 1; i <= n; i++) {
        if (dist[i] == INF)
            cout << -1 << '\n';
        else
            cout << dist[i] << '\n';
    }

    return 0;
}
```

## 1.3 Bellman-Ford

```cpp
#include <bits/stdc++.h>
using namespace std;

using ll = long long;

const ll INF = (1LL << 60);

struct Edge {
    int u, v;
    ll w;
};

vector<Edge> edges;
vector<ll> dist;

bool bellmanFord(int n, int s) {
    dist.assign(n + 1, INF);
    dist[s] = 0;

    // 最多进行 n - 1 轮松弛
    for (int i = 1; i <= n - 1; i++) {
        bool updated = false;

        for (auto [u, v, w] : edges) {
            if (dist[u] == INF) continue;

            if (dist[v] > dist[u] + w) {
                dist[v] = dist[u] + w;
                updated = true;
            }
        }

        // 本轮没有任何更新，说明最短路已经确定
        if (!updated) break;
    }

    // 再进行一轮松弛
    // 如果还能更新，说明存在从起点可达的负环
    for (auto [u, v, w] : edges) {
        if (dist[u] == INF) continue;

        if (dist[v] > dist[u] + w) {
            return false;
        }
    }

    return true;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m, s;
    cin >> n >> m >> s;

    for (int i = 0; i < m; i++) {
        int u, v;
        ll w;
        cin >> u >> v >> w;

        // 有向图
        edges.push_back({u, v, w});

        // 如果是无向图，需要加入反向边
        // edges.push_back({v, u, w});
    }

    bool ok = bellmanFord(n, s);

    if (!ok) {
        cout << "Negative cycle exists\n";
        return 0;
    }

    for (int i = 1; i <= n; i++) {
        if (dist[i] == INF)
            cout << -1 << '\n';
        else
            cout << dist[i] << '\n';
    }

    return 0;
}
```

## SPFA
```cpp
#include <bits/stdc++.h>
using namespace std;

using ll = long long;

const ll INF = (1LL << 60);
const int MAXN = 200005;

struct Edge {
    int to;
    ll w;
};

vector<Edge> g[MAXN];

ll dist[MAXN];
bool inq[MAXN];
int cnt[MAXN];

// 返回 false 表示存在从起点可达的负环
bool spfa(int s, int n) {
    fill(dist + 1, dist + n + 1, INF);
    fill(inq + 1, inq + n + 1, false);
    fill(cnt + 1, cnt + n + 1, 0);

    queue<int> q;

    dist[s] = 0;
    q.push(s);
    inq[s] = true;

    while (!q.empty()) {
        int u = q.front();
        q.pop();

        inq[u] = false;

        for (auto [v, w] : g[u]) {
            if (dist[v] > dist[u] + w) {
                dist[v] = dist[u] + w;

                if (!inq[v]) {
                    q.push(v);
                    inq[v] = true;

                    // 记录入队次数
                    cnt[v]++;

                    // 入队达到 n 次，说明存在负环
                    if (cnt[v] >= n) {
                        return false;
                    }
                }
            }
        }
    }

    return true;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m, s;
    cin >> n >> m >> s;

    for (int i = 0; i < m; i++) {
        int u, v;
        ll w;

        cin >> u >> v >> w;

        // 有向图
        g[u].push_back({v, w});

        // 无向图：
        // g[v].push_back({u, w});
    }

    if (!spfa(s, n)) {
        cout << "Negative cycle exists\n";
        return 0;
    }

    for (int i = 1; i <= n; i++) {
        if (dist[i] == INF)
            cout << -1 << '\n';
        else
            cout << dist[i] << '\n';
    }

    return 0;
}
```
