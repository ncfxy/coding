# 欧拉函数

## 求某个数的欧拉函数

```cpp
#include<vector>

using namespace std;

// ------------------------------------------------------------
// 预处理 sqrt(1e7) 范围内的质数
// ------------------------------------------------------------
vector<int> getPrimes() {
    const int N = 3200;

    vector<bool> isPrime(N + 1, true);
    isPrime[0] = isPrime[1] = false;

    for (int i = 2; i * i <= N; ++i) {
        if (!isPrime[i]) continue;

        for (int j = i * i; j <= N; j += i) {
            isPrime[j] = false;
        }
    }

    vector<int> primes;

    for (int i = 2; i <= N; ++i) {
        if (isPrime[i]) {
            primes.push_back(i);
        }
    }

    return primes;
}

vector<int> primes;

// ------------------------------------------------------------
// Euler Phi
// ------------------------------------------------------------
int phi(int x) {
    if (x == 1) {
        return 1;
    }

    int n = x;
    int ans = x;

    for (int p : primes) {
        if (1LL * p * p > n) {
            break;
        }

        if (n % p == 0) {
            ans = ans / p * (p - 1);

            while (n % p == 0) {
                n /= p;
            }
        }
    }

    // 最后剩下一个大于 sqrt(original x) 的质因子
    if (n > 1) {
        ans = ans / n * (n - 1);
    }

    return ans;
}
```

## 批量得到欧拉函数
```cpp
#include <cstdio>
#include <cstring>

using namespace std;

#define N 10000000
int main() {
    int phi[N +1];
    bool prime[N +1];
    memset(prime, true, sizeof(prime));
    for(int i = 2;i * i <=N; i++) {
        if(prime[i]) {
            for(int j = i * i;j <= N;j += i) {
                prime[j] = false;
            }
        }
    } // 求出N以内的素数
    for(int i = 1; i <= N; i++) {
        phi[i] = i;
    }
    for(int i = 2; i <= N; i++) {
        if(prime[i]) {
            for(int j = i; j <= N; j += i) {
                phi[j] = phi[j] / i * (i - 1);
            }
        }
    }// N 以内的所有欧拉函数值
}
```
