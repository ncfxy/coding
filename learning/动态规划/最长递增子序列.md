# 最长递增子序列

## 1.1 动态规划 O(n^2)
- dp[i] = 以 a[i] 结尾的最长递增子序列长度
- 状态转移: \(dp[i]=\max(dp[i],dp[j]+1),\quad j<i,\ a[j]<a[i]\)

```cpp
#include <bits/stdc++.h>
using namespace std;

int LIS(const vector<int> & a) {
    int n = a.size();
    vector<int> dp(n, 1);

    int ans = 0;

    for(int i = 0;i < n;i++) {
        for(int j = 0;j < i;j++) {
            if(a[j] < a[i]) {
                dp[i] = max(dp[i], dp[j] + 1);
            }
        }
        ans = max(ans, dp[i]);
    }
    return ans;
}

int main() {
    vector<int> a = {3, 1, 2, 5, 4, 7};

    cout << LIS(a) << endl;

    return 0;
}
```

## 1.2 贪心+二分查找 O(n logn)
- d 数组中 d[len-1] 维护长度为len的递增子序列中，最小可能得结尾元素
```cpp
#include <bits/stdc++.h>
using namespace std;

int LIS(const vector<int> & a) {
    vector<int> d;

    for(int x: a) {
        auto it = lower_bound(d.begin(), d.end(), x);

        if(it == d.end()) {
            d.push_back(x);
        } else {
            *it = x;
        }
    }
    return d.size();
}

int main() {
    vector<int> a = {3, 1, 2, 5, 4, 7};

    cout << LIS(a) << endl;

    return 0;
}
```
